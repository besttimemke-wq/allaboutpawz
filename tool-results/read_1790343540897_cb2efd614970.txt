     1→'use client';
     2→
     3→import { useEffect, useState, useCallback, useRef } from 'react';
     4→import { cn } from '@/lib/utils';
     5→import {
     6→  Search, ShoppingCart, User, Package, Scissors, Repeat, RefreshCw,
     7→  Plus, Minus, X, DollarSign, Receipt, Lock, Tag, Percent, Wallet,
     8→  Gift, Check, Pause, Trash2, CreditCard,
     9→} from 'lucide-react';
    10→
    11→type CatalogItem = {
    12→  id: string; sku: string; name: string;
    13→  itemType: 'product' | 'service' | 'subscription';
    14→  price: number; compareAtPrice: number | null; stock: number | null;
    15→  categoryName?: string;
    16→};
    17→
    18→type Category = { id: string | null; name: string; itemCount: number };
    19→type PaymentMethod = { id: string; code: string; name: string; methodType: string };
    20→type Register = {
    21→  id: string; registerNumber: string; name: string;
    22→  activeSession: { id: string; openingCash: number; expectedCash: number; openedAt: string } | null;
    23→};
    24→
    25→type CartLine = {
    26→  key: string; catalogItemId?: string; serviceId?: string; subscriptionPlanId?: string;
    27→  description: string; quantity: number; unitPrice: number; discountAmount: number;
    28→  itemType: 'product' | 'service' | 'subscription';
    29→};
    30→
    31→type TodaySummary = {
    32→  salesCount: number; grossSales: number; discounts: number;
    33→  tax: number; netSales: number; cashSales: number; cardSales: number;
    34→};
    35→
    36→// ---- localStorage cart backup (survives browser crash/refresh) ----
    37→const CART_STORAGE_KEY = (sessionId: string) => `pos-cart-${sessionId}`;
    38→
    39→export default function PosPage() {
    40→  const [catalog, setCatalog] = useState<CatalogItem[]>([]);
    41→  const [categories, setCategories] = useState<Category[]>([]);
    42→  const [paymentMethods, setPaymentMethods] = useState<PaymentMethod[]>([]);
    43→  const [registers, setRegisters] = useState<Register[]>([]);
    44→  const [summary, setSummary] = useState<TodaySummary | null>(null);
    45→  const [loading, setLoading] = useState(true);
    46→  const [search, setSearch] = useState('');
    47→  const [activeCategory, setActiveCategory] = useState<string | null>(null);
    48→  const [cart, setCart] = useState<CartLine[]>([]);
    49→  const [heldCarts, setHeldCarts] = useState<CartLine[][]>([]);
    50→  const [customerEmail, setCustomerEmail] = useState('');
    51→  const [showPayment, setShowPayment] = useState(false);
    52→  const [showOpenDrawer, setShowOpenDrawer] = useState(false);
    53→  const [showQuickAdd, setShowQuickAdd] = useState(false);
    54→  const [openingCash, setOpeningCash] = useState('100.00');
    55→  const [busy, setBusy] = useState(false);
    56→  const [lastSale, setLastSale] = useState<any>(null);
    57→  const [error, setError] = useState<string | null>(null);
    58→
    59→  const searchInputRef = useRef<HTMLInputElement>(null);
    60→  const activeRegister = registers[0];
    61→  const activeSession = activeRegister?.activeSession;
    62→
    63→  const load = useCallback(async () => {
    64→    try {
    65→      const res = await fetch('/api/admin/pos');
    66→      if (!res.ok) return;
    67→      const data = await res.json();
    68→      setCatalog(data.catalog || []);
    69→      setCategories(data.categories || []);
    70→      setPaymentMethods(data.paymentMethods || []);
    71→      setRegisters(data.registers || []);
    72→      setSummary(data.todaySummary || null);
    73→    } catch {} finally { setLoading(false); }
    74→  }, []);
    75→
    76→  useEffect(() => { load(); }, [load]);
    77→
    78→  // ---- localStorage cart backup ----
    79→  // Mirror the active cart to localStorage keyed by register_session_id.
    80→  // If the browser crashes or the cashier hits refresh, the cart survives.
    81→  useEffect(() => {
    82→    if (activeSession?.id) {
    83→      const saved = localStorage.getItem(CART_STORAGE_KEY(activeSession.id));
    84→      if (saved && cart.length === 0) {
    85→        try { setCart(JSON.parse(saved)); } catch {}
    86→      }
    87→    }
    88→  }, [activeSession?.id]);
    89→
    90→  useEffect(() => {
    91→    if (activeSession?.id) {
    92→      localStorage.setItem(CART_STORAGE_KEY(activeSession.id), JSON.stringify(cart));
    93→    }
    94→  }, [cart, activeSession?.id]);
    95→
    96→  // ---- keyboard shortcut: focus search on "/" ----
    97→  useEffect(() => {
    98→    const handler = (e: KeyboardEvent) => {
    99→      if (e.key === '/' && document.activeElement?.tagName !== 'INPUT') {
   100→        e.preventDefault();
   101→        searchInputRef.current?.focus();
   102→      }
   103→    };
   104→    document.addEventListener('keydown', handler);
   105→    return () => document.removeEventListener('keydown', handler);
   106→  }, []);
   107→
   108→  const filtered = catalog.filter((c) => {
   109→    if (activeCategory && activeCategory !== 'All Items' && c.categoryName !== activeCategory) return false;
   110→    if (search) {
   111→      const q = search.toLowerCase();
   112→      return c.name.toLowerCase().includes(q) || c.sku.toLowerCase().includes(q);
   113→    }
   114→    return true;
   115→  });
   116→
   117→  const addToCart = (item: CatalogItem) => {
   118→    setCart((prev) => {
   119→      const key = `${item.itemType}-${item.id}`;
   120→      const existing = prev.find((l) => l.key === key);
   121→      if (existing) return prev.map((l) => l.key === key ? { ...l, quantity: l.quantity + 1 } : l);
   122→      return [...prev, {
   123→        key, catalogItemId: item.itemType === 'product' ? item.id : undefined,
   124→        serviceId: item.itemType === 'service' ? item.id : undefined,
   125→        subscriptionPlanId: item.itemType === 'subscription' ? item.id : undefined,
   126→        description: item.name, quantity: 1, unitPrice: item.price, discountAmount: 0, itemType: item.itemType,
   127→      }];
   128→    });
   129→    setSearch('');
   130→    searchInputRef.current?.focus();
   131→  };
   132→
   133→  const updateQty = (key: string, delta: number) => {
   134→    setCart((prev) => prev.map((l) => l.key === key ? { ...l, quantity: Math.max(1, l.quantity + delta) } : l));
   135→  };
   136→  const updatePrice = (key: string, price: number) => {
   137→    setCart((prev) => prev.map((l) => l.key === key ? { ...l, unitPrice: price } : l));
   138→  };
   139→  const removeLine = (key: string) => setCart((prev) => prev.filter((l) => l.key !== key));
   140→  const clearCart = () => { setCart([]); setError(null); };
   141→  const holdCart = () => { if (cart.length > 0) { setHeldCarts([...heldCarts, cart]); setCart([]); } };
   142→  const resumeCart = (idx: number) => { setCart(heldCarts[idx]); setHeldCarts(heldCarts.filter((_, i) => i !== idx)); };
   143→
   144→  const subtotal = cart.reduce((s, l) => s + (l.unitPrice * l.quantity) - l.discountAmount, 0);
   145→  const discountTotal = cart.reduce((s, l) => s + l.discountAmount, 0);
   146→  const taxRate = 0.0925;
   147→  const taxTotal = Math.round((subtotal - discountTotal) * taxRate * 100) / 100;
   148→  const total = subtotal - discountTotal + taxTotal;
   149→
   150→  const openDrawer = async () => {
   151→    if (!activeRegister || busy) return;
   152→    setBusy(true); setError(null);
   153→    try {
   154→      const res = await fetch('/api/admin/pos', {
   155→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({ action: 'open_register', registerId: activeRegister.id, openingCash: Number(openingCash) }),
   157→      });
   158→      if (!res.ok) { const d = await res.json(); setError(d.error || 'Failed'); setBusy(false); return; }
   159→      setShowOpenDrawer(false); load();
   160→    } catch { setError('Failed'); } finally { setBusy(false); }
   161→  };
   162→
   163→  const completeSale = async (payments: any[]) => {
   164→    if (!activeSession || busy) return;
   165→    setBusy(true); setError(null);
   166→    try {
   167→      // Generate a client-side idempotency key — blocks double-submission.
   168→      const idempotencyKey = crypto.randomUUID();
   169→      const res = await fetch('/api/admin/pos', {
   170→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   171→        body: JSON.stringify({
   172→          action: 'complete_sale',
   173→          idempotencyKey,
   174→          registerSessionId: activeSession.id,
   175→          lines: cart.map((l) => ({
   176→            catalogItemId: l.catalogItemId, serviceId: l.serviceId,
   177→            subscriptionPlanId: l.subscriptionPlanId, description: l.description,
   178→            quantity: l.quantity, unitPrice: l.unitPrice, discountAmount: l.discountAmount,
   179→            itemType: l.itemType,
   180→          })),
   181→          discountTotal, taxTotal, payments,
   182→        }),
   183→      });
   184→      const data = await res.json();
   185→      if (!res.ok) { setError(data.error || 'Sale failed'); setBusy(false); return; }
   186→      setLastSale(data.sale);
   187→      setCart([]);
   188→      if (activeSession?.id) localStorage.removeItem(CART_STORAGE_KEY(activeSession.id));
   189→      setShowPayment(false); load();
   190→    } catch { setError('Sale failed'); } finally { setBusy(false); }
   191→  };
   192→
   193→  // ---- thermal receipt printing (raw ESC/POS via Web Serial) ----
   194→  const printReceipt = async (receiptRaw: string) => {
   195→    // Try Web Serial API (Chrome) for direct thermal printer connection.
   196→    // Falls back to a new window with pre-formatted text if Web Serial
   197→    // isn't available (non-Chrome browsers, no printer connected).
   198→    if ('serial' in navigator) {
   199→      try {
   200→        const port = await (navigator as any).serial.requestPort();
   201→        await port.open({ baudRate: 9600 });
   202→        const writer = port.writable.getWriter();
   203→        await writer.write(new TextEncoder().encode(receiptRaw));
   204→        writer.releaseLock();
   205→        await port.close();
   206→        return;
   207→      } catch (e) {
   208→        // User cancelled or printer not available — fall through to window print
   209→      }
   210→    }
   211→    // Fallback: open a plain-text window (no HTML margins, monospace)
   212→    const w = window.open('', '_blank', 'width=400,height=600');
   213→    if (w) {
   214→      w.document.write(`<pre style="font-family:monospace;font-size:11px;margin:0;padding:8px;">${receiptRaw.replace(/</g, '&lt;')}</pre>`);
   215→      w.document.close();
   216→      w.print();
   217→    }
   218→  };
   219→
   220→  if (loading) return <div className="p-6 text-muted-foreground">Loading POS…</div>;
   221→
   222→  // ---- No active session → open drawer screen ----
   223→  if (!activeSession) {
   224→    return (
   225→      <div className="flex flex-col items-center justify-center min-h-full bg-background p-6">
   226→        <div className="w-full max-w-md space-y-6">
   227→          <div className="text-center">
   228→            <Lock className="size-12 mx-auto text-muted-foreground/40" />
   229→            <h1 className="text-2xl font-semibold tracking-tight mt-4">Register Closed</h1>
   230→            <p className="text-[13px] text-muted-foreground mt-1">
   231→              {activeRegister?.name || 'Register #1'} — open the drawer to start making sales.
   232→            </p>
   233→          </div>
   234→          <div className="bg-card border border-border rounded-xl p-6 space-y-4">
   235→            <div>
   236→              <label className="block text-[12px] font-medium text-foreground mb-1.5">Opening Cash Float</label>
   237→              <div className="relative">
   238→                <DollarSign className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   239→                <input type="number" step="0.01" value={openingCash} onChange={(e) => setOpeningCash(e.target.value)}
   240→                  className="w-full border border-border rounded-lg pl-9 pr-3 py-2.5 text-[15px] focus:outline-none focus:border-foreground" />
   241→              </div>
   242→            </div>
   243→            <button onClick={openDrawer} disabled={busy}
   244→              className="w-full py-3 bg-ink text-white rounded-lg text-[13px] font-semibold tracking-wider uppercase hover:opacity-90 disabled:opacity-60 cursor-pointer">
   245→              {busy ? 'Opening…' : 'Open Register'}
   246→            </button>
   247→            {error && <p className="text-[12px] text-destructive text-center">{error}</p>}
   248→          </div>
   249→        </div>
   250→      </div>
   251→    );
   252→  }
   253→
   254→  return (
   255→    <div className="flex h-full bg-background overflow-hidden">
   256→      {/* =========================================== */}
   257→      {/* LEFT: Catalog & Inputs (65% width) */}
   258→      {/* =========================================== */}
   259→      <div className="flex-1 flex flex-col overflow-hidden border-r border-border" style={{ flexBasis: '65%' }}>
   260→        {/* Search/barcode bar */}
   261→        <div className="p-3 border-b border-border">
   262→          <div className="flex items-center gap-2">
   263→            <div className="flex-1 relative">
   264→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   265→              <input
   266→                ref={searchInputRef}
   267→                type="text" value={search} onChange={(e) => setSearch(e.target.value)}
   268→                placeholder="Search or scan barcode… (press / to focus)"
   269→                className="w-full border border-border rounded-lg pl-9 pr-3 py-2.5 text-[13px] focus:outline-none focus:border-ink"
   270→                autoFocus
   271→              />
   272→            </div>
   273→            <button onClick={load} className="p-2.5 rounded-lg border border-border hover:bg-muted cursor-pointer" title="Refresh">
   274→              <RefreshCw className="size-4 text-muted-foreground" />
   275→            </button>
   276→            <button onClick={() => setShowQuickAdd(true)} className="px-3 py-2.5 rounded-lg bg-ink text-white text-[12px] font-semibold hover:opacity-90 cursor-pointer flex items-center gap-1.5" title="Quick add item">
   277→              <Plus className="size-4" /> Add Item
   278→            </button>
   279→          </div>
   280→        </div>
   281→
   282→        {/* Category navigation */}
   283→        <div className="px-3 py-2 border-b border-border overflow-x-auto custom-scrollbar">
   284→          <div className="flex items-center gap-1">
   285→            {categories.map((cat) => (
   286→              <button
   287→                key={cat.name} onClick={() => setActiveCategory(cat.name === 'All Items' ? null : cat.name)}
   288→                className={cn(
   289→                  'px-3 py-1.5 rounded-full text-[12px] font-medium whitespace-nowrap transition-colors cursor-pointer',
   290→                  (activeCategory === null && cat.name === 'All Items') || activeCategory === cat.name
   291→                    ? 'bg-ink text-white' : 'bg-muted/50 text-muted-foreground hover:bg-muted',
   292→                )}
   293→              >
   294→                {cat.name} <span className="opacity-60">({cat.itemCount})</span>
   295→              </button>
   296→            ))}
   297→          </div>
   298→        </div>
   299→
   300→        {/* Product quick-keys grid */}
   301→        <div className="flex-1 overflow-y-auto custom-scrollbar p-3">
   302→          {filtered.length === 0 ? (
   303→            <div className="text-center py-12 text-muted-foreground text-[13px]">No items found.</div>
   304→          ) : (
   305→            <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 xl:grid-cols-5 gap-2.5">
   306→              {filtered.map((item) => (
   307→                <button
   308→                  key={item.id} onClick={() => addToCart(item)}
   309→                  className="flex flex-col items-start p-3 bg-card border border-border rounded-lg hover:border-ink hover:shadow-sm transition-all text-left cursor-pointer"
   310→                >
   311→                  <div className="flex items-center gap-1 mb-1">
   312→                    {item.itemType === 'product' && <Package className="size-3.5 text-muted-foreground" />}
   313→                    {item.itemType === 'service' && <Scissors className="size-3.5 text-muted-foreground" />}
   314→                    {item.itemType === 'subscription' && <Repeat className="size-3.5 text-muted-foreground" />}
   315→                    <span className="text-[9px] font-bold uppercase tracking-wider text-muted-foreground">{item.itemType}</span>
   316→                  </div>
   317→                  <span className="text-[13px] font-medium leading-tight line-clamp-2">{item.name}</span>
   318→                  <div className="flex items-center justify-between w-full mt-2">
   319→                    <span className="text-[14px] font-bold">${item.price.toFixed(2)}</span>
   320→                    {item.stock != null && item.stock > 0 && <span className="text-[10px] text-muted-foreground">{item.stock}</span>}
   321→                  </div>
   322→                </button>
   323→              ))}
   324→            </div>
   325→          )}
   326→        </div>
   327→      </div>
   328→
   329→      {/* =========================================== */}
   330→      {/* RIGHT: Cart & Summary (35% width) */}
   331→      {/* =========================================== */}
   332→      <div className="flex flex-col bg-card overflow-hidden" style={{ flexBasis: '35%', maxWidth: '480px' }}>
   333→        {/* Cart header */}
   334→        <div className="p-3 border-b border-border">
   335→          <div className="flex items-center justify-between mb-2">
   336→            <h2 className="text-[14px] font-semibold flex items-center gap-2">
   337→              <ShoppingCart className="size-4" /> Current Ticket
   338→            </h2>
   339→            {activeSession && (
   340→              <span className="text-[10px] font-medium text-success bg-success/10 px-2 py-1 rounded-full">
   341→                Drawer ${activeSession.openingCash.toFixed(2)}
   342→              </span>
   343→            )}
   344→          </div>
   345→          {/* Customer */}
   346→          <div className="relative">
   347→            <User className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   348→            <input type="email" value={customerEmail} onChange={(e) => setCustomerEmail(e.target.value)}
   349→              placeholder="Customer email (optional)…"
   350→              className="w-full border border-border rounded-lg pl-8 pr-3 py-1.5 text-[12px] focus:outline-none focus:border-ink" />
   351→          </div>
   352→        </div>
   353→
   354→        {/* Held carts */}
   355→        {heldCarts.length > 0 && (
   356→          <div className="px-3 py-2 border-b border-border bg-muted/30">
   357→            <p className="text-[10px] font-bold uppercase tracking-wider text-muted-foreground mb-1">Held Tickets ({heldCarts.length})</p>
   358→            <div className="flex gap-1 flex-wrap">
   359→              {heldCarts.map((hc, i) => (
   360→                <button key={i} onClick={() => resumeCart(i)}
   361→                  className="px-2 py-1 rounded text-[11px] bg-card border border-border hover:bg-muted cursor-pointer">
   362→                  <Pause className="size-2.5 inline mr-1" />{hc.length} items
   363→                </button>
   364→              ))}
   365→            </div>
   366→          </div>
   367→        )}
   368→
   369→        {/* Cart lines */}
   370→        <div className="flex-1 overflow-y-auto custom-scrollbar">
   371→          {cart.length === 0 ? (
   372→            <div className="flex flex-col items-center justify-center h-full py-12 text-muted-foreground">
   373→              <ShoppingCart className="size-10 mb-3 opacity-30" />
   374→              <p className="text-[13px]">Scan or tap items to start</p>
   375→            </div>
   376→          ) : (
   377→            <div className="divide-y divide-border">
   378→              {cart.map((line) => (
   379→                <div key={line.key} className="p-2.5 space-y-1.5">
   380→                  <div className="flex items-start justify-between gap-2">
   381→                    <span className="text-[12px] font-medium flex-1">{line.description}</span>
   382→                    <button onClick={() => removeLine(line.key)} className="text-muted-foreground hover:text-destructive cursor-pointer">
   383→                      <X className="size-3.5" />
   384→                    </button>
   385→                  </div>
   386→                  <div className="flex items-center gap-2">
   387→                    <div className="flex items-center gap-1">
   388→                      <button onClick={() => updateQty(line.key, -1)} className="p-0.5 rounded border border-border hover:bg-muted cursor-pointer"><Minus className="size-3" /></button>
   389→                      <span className="w-7 text-center text-[12px] font-medium">{line.quantity}</span>
   390→                      <button onClick={() => updateQty(line.key, 1)} className="p-0.5 rounded border border-border hover:bg-muted cursor-pointer"><Plus className="size-3" /></button>
   391→                    </div>
   392→                    <div className="relative">
   393→                      <DollarSign className="absolute left-1.5 top-1/2 -translate-y-1/2 size-2.5 text-muted-foreground" />
   394→                      <input type="number" step="0.01" value={line.unitPrice} onChange={(e) => updatePrice(line.key, Number(e.target.value))}
   395→                        className="w-16 border border-border rounded pl-5 pr-1 py-0.5 text-[11px] focus:outline-none focus:border-ink" />
   396→                    </div>
   397→                    <span className="ml-auto text-[12px] font-semibold">${(line.unitPrice * line.quantity - line.discountAmount).toFixed(2)}</span>
   398→                  </div>
   399→                </div>
   400→              ))}
   401→            </div>
   402→          )}
   403→        </div>
   404→
   405→        {/* Order totals + action buttons */}
   406→        {cart.length > 0 && (
   407→          <div className="border-t border-border p-3 space-y-2.5">
   408→            <div className="space-y-0.5 text-[12px]">
   409→              <div className="flex justify-between text-muted-foreground"><span>Subtotal</span><span>${subtotal.toFixed(2)}</span></div>
   410→              {discountTotal > 0 && <div className="flex justify-between text-muted-foreground"><span>Discounts</span><span>−${discountTotal.toFixed(2)}</span></div>}
   411→              <div className="flex justify-between text-muted-foreground"><span>Tax (9.25%)</span><span>${taxTotal.toFixed(2)}</span></div>
   412→              <div className="flex justify-between text-[16px] font-bold pt-1 border-t border-border"><span>Total</span><span>${total.toFixed(2)}</span></div>
   413→            </div>
   414→            {/* Action buttons */}
   415→            <div className="flex gap-1.5">
   416→              <button onClick={clearCart} className="flex-1 py-2 border border-border rounded-lg text-[11px] font-medium hover:bg-muted cursor-pointer flex items-center justify-center gap-1">
   417→                <Trash2 className="size-3" /> Clear
   418→              </button>
   419→              <button onClick={holdCart} className="flex-1 py-2 border border-border rounded-lg text-[11px] font-medium hover:bg-muted cursor-pointer flex items-center justify-center gap-1">
   420→                <Pause className="size-3" /> Hold
   421→              </button>
   422→            </div>
   423→            <button onClick={() => setShowPayment(true)}
   424→              className="w-full py-3 bg-ink text-white rounded-lg text-[14px] font-semibold tracking-wider uppercase hover:opacity-90 cursor-pointer flex items-center justify-center gap-2">
   425→              <Wallet className="size-4" /> Charge ${total.toFixed(2)}
   426→            </button>
   427→          </div>
   428→        )}
   429→
   430→        {/* Today's summary */}
   431→        {summary && (
   432→          <div className="border-t border-border p-2.5 bg-muted/30">
   433→            <div className="grid grid-cols-3 gap-2 text-center">
   434→              <div><p className="text-[15px] font-bold">{summary.salesCount}</p><p className="text-[9px] text-muted-foreground uppercase">Sales</p></div>
   435→              <div><p className="text-[15px] font-bold">${summary.netSales.toFixed(0)}</p><p className="text-[9px] text-muted-foreground uppercase">Net</p></div>
   436→              <div><p className="text-[15px] font-bold">${(summary.cashSales + summary.cardSales).toFixed(0)}</p><p className="text-[9px] text-muted-foreground uppercase">Tendered</p></div>
   437→            </div>
   438→          </div>
   439→        )}
   440→      </div>
   441→
   442→      {/* Payment drawer */}
   443→      {showPayment && (
   444→        <PaymentDrawer
   445→          total={total} subtotal={subtotal} taxTotal={taxTotal} discountTotal={discountTotal}
   446→          paymentMethods={paymentMethods} onComplete={completeSale} onClose={() => setShowPayment(false)}
   447→          busy={busy} error={error}
   448→        />
   449→      )}
   450→
   451→      {/* Sale success + receipt printing */}
   452→      {lastSale && (
   453→        <div className="fixed inset-0 z-50 flex items-center justify-center bg-foreground/90 p-4">
   454→          <div className="w-full max-w-sm bg-card rounded-2xl border border-border p-6 space-y-4 text-center">
   455→            <Receipt className="size-12 mx-auto text-success" />
   456→            <h3 className="text-[18px] font-semibold">Sale Complete</h3>
   457→            <div className="space-y-1 text-[13px]">
   458→              <p className="text-muted-foreground">Receipt #{lastSale.receiptNumber}</p>
   459→              <p className="text-[20px] font-bold">${lastSale.total.toFixed(2)}</p>
   460→              {lastSale.changeDue > 0 && <p className="text-[14px] font-semibold text-success">Change: ${lastSale.changeDue.toFixed(2)}</p>}
   461→              {lastSale.journalEntryId && <p className="text-[10px] text-muted-foreground">GL Entry: {lastSale.journalEntryId.slice(0,8)}</p>}
   462→            </div>
   463→            <div className="flex gap-2">
   464→              <button onClick={() => printReceipt(lastSale.receiptRaw)}
   465→                className="flex-1 py-2.5 border border-border rounded-lg text-[13px] font-semibold hover:bg-muted cursor-pointer flex items-center justify-center gap-1.5">
   466→                <Receipt className="size-3.5" /> Print
   467→              </button>
   468→              <button onClick={() => setLastSale(null)}
   469→                className="flex-1 py-2.5 bg-ink text-white rounded-lg text-[13px] font-semibold cursor-pointer">
   470→                New Sale
   471→              </button>
   472→            </div>
   473→          </div>
   474→        </div>
   475→      )}
   476→
   477→      {/* Quick Add Item modal — cashier creates a new item on the fly */}
   478→      {showQuickAdd && (
   479→        <QuickAddItem
   480→          onClose={() => setShowQuickAdd(false)}
   481→          onCreated={(item) => {
   482→            // Add the new item to the cart immediately
   483→            setCart((prev) => {
   484→              const key = `product-${item.id}`;
   485→              return [...prev, {
   486→                key, catalogItemId: item.id, description: item.name,
   487→                quantity: 1, unitPrice: item.price, discountAmount: 0,
   488→                itemType: 'product' as const,
   489→              }];
   490→            });
   491→            setShowQuickAdd(false);
   492→            load(); // refresh catalog
   493→          }}
   494→        />
   495→      )}
   496→    </div>
   497→  );
   498→}

   500→// ---- Payment Drawer (cash/change, check, gift card, card) ----
   501→function PaymentDrawer({
   502→  total, subtotal, taxTotal, discountTotal, paymentMethods, onComplete, onClose, busy, error,
   503→}: {
   504→  total: number; subtotal: number; taxTotal: number; discountTotal: number;
   505→  paymentMethods: PaymentMethod[];
   506→  onComplete: (payments: any[]) => void;
   507→  onClose: () => void; busy: boolean; error: string | null;
   508→}) {
   509→  const [selectedMethod, setSelectedMethod] = useState<string>('');
   510→  const [amount, setAmount] = useState(total.toFixed(2));
   511→  const [tip, setTip] = useState('0.00');
   512→  const [payments, setPayments] = useState<any[]>([]);
   513→  const [giftCardNumber, setGiftCardNumber] = useState('');
   514→  const [giftCardBalance, setGiftCardBalance] = useState<number | null>(null);
   515→  const [checkRef, setCheckRef] = useState('');
   516→
   517→  useEffect(() => {
   518→    // eslint-disable-next-line react-hooks/set-state-in-effect
   519→    if (paymentMethods.length > 0 && !selectedMethod) setSelectedMethod(paymentMethods[0].id);
   520→  }, [paymentMethods, selectedMethod]);
   521→
   522→  const remaining = Math.max(0, total - payments.reduce((s, p) => s + p.amount, 0));
   523→  const selectedPM = paymentMethods.find(m => m.id === selectedMethod);
   524→  const isCash = selectedPM?.methodType === 'cash';
   525→  const isCheck = selectedPM?.methodType === 'check';
   526→  const isGiftCard = selectedPM?.code === 'STORE_CREDIT';
   527→  const changeDue = Math.max(0, Number(amount) - remaining);
   528→
   529→  const checkGiftCard = async () => {
   530→    const res = await fetch('/api/admin/pos', {
   531→      method: 'POST', headers: { 'Content-Type': 'application/json' },
   532→      body: JSON.stringify({ action: 'query_gift_card', cardNumber: giftCardNumber }),
   533→    });
   534→    const data = await res.json();
   535→    if (res.ok) setGiftCardBalance(data.card.balance);
   536→    else { setGiftCardBalance(null); alert(data.error || 'Gift card not found'); }
   537→  };
   538→
   539→  const addPayment = () => {
   540→    const amt = Number(amount);
   541→    if (!selectedMethod || amt <= 0) return;
   542→    if (isGiftCard && giftCardBalance != null && amt > giftCardBalance) {
   543→      alert(`Gift card balance is only $${giftCardBalance.toFixed(2)}`);
   544→      return;
   545→    }
   546→    setPayments([...payments, {
   547→      paymentMethodId: selectedMethod, amount: amt, tipAmount: Number(tip) || 0,
   548→      giftCardNumber: isGiftCard ? giftCardNumber : undefined,
   549→      checkReference: isCheck ? checkRef : undefined,
   550→    }]);
   551→    setAmount(remaining.toFixed(2)); setTip('0.00');
   552→    if (isGiftCard) { setGiftCardNumber(''); setGiftCardBalance(null); }
   553→    if (isCheck) setCheckRef('');
   554→  };
   555→
   556→  const grouped = paymentMethods.reduce((acc, m) => {
   557→    (acc[m.methodType] = acc[m.methodType] || []).push(m);
   558→    return acc;
   559→  }, {} as Record<string, PaymentMethod[]>);
   560→
   561→  return (
   562→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-foreground/90 p-4">
   563→      <div className="w-full max-w-lg bg-card rounded-2xl border border-border shadow-2xl flex flex-col max-h-[90vh]">
   564→        <div className="flex items-center justify-between p-5 border-b border-border">
   565→          <h3 className="text-[16px] font-semibold">Payment</h3>
   566→          <button onClick={onClose} className="p-1 rounded-lg text-muted-foreground hover:text-foreground cursor-pointer"><X className="size-5" /></button>
   567→        </div>
   568→
   569→        <div className="flex-1 overflow-y-auto p-5 space-y-4">
   570→          {/* Totals */}
   571→          <div className="bg-muted/40 rounded-lg p-3 space-y-0.5 text-[12px]">
   572→            <div className="flex justify-between text-muted-foreground"><span>Subtotal</span><span>${subtotal.toFixed(2)}</span></div>
   573→            {discountTotal > 0 && <div className="flex justify-between text-muted-foreground"><span>Discounts</span><span>−${discountTotal.toFixed(2)}</span></div>}
   574→            <div className="flex justify-between text-muted-foreground"><span>Tax</span><span>${taxTotal.toFixed(2)}</span></div>
   575→            <div className="flex justify-between text-[17px] font-bold pt-1 border-t border-border"><span>Total Due</span><span>${total.toFixed(2)}</span></div>
   576→          </div>
   577→
   578→          {/* Tendered so far */}
   579→          {payments.length > 0 && (
   580→            <div className="space-y-1">
   581→              {payments.map((p, i) => {
   582→                const m = paymentMethods.find(m => m.id === p.paymentMethodId);
   583→                return <div key={i} className="flex justify-between text-[12px]"><span>{m?.name}</span><span>${p.amount.toFixed(2)}</span></div>;
   584→              })}
   585→              <div className="flex justify-between text-[13px] font-semibold pt-1 border-t border-border">
   586→                <span className="text-muted-foreground">Remaining</span>
   587→                <span className={remaining > 0 ? 'text-destructive' : 'text-success'}>${remaining.toFixed(2)}</span>
   588→              </div>
   589→            </div>
   590→          )}
   591→
   592→          {/* Payment method selector */}
   593→          <div className="space-y-2">
   594→            {Object.entries(grouped).map(([type, methods]) => (
   595→              <div key={type}>
   596→                <p className="text-[10px] text-muted-foreground/70 capitalize mb-1">{type}</p>
   597→                <div className="grid grid-cols-3 gap-2">
   598→                  {methods.map((m) => (
   599→                    <button key={m.id} onClick={() => setSelectedMethod(m.id)}
   600→                      className={cn('py-2 px-2 rounded-lg text-[12px] font-medium border transition-colors cursor-pointer',
   601→                        selectedMethod === m.id ? 'bg-ink text-white border-ink' : 'border-border hover:bg-muted')}>
   602→                      {m.name}
   603→                    </button>
   604→                  ))}
   605→                </div>
   606→              </div>
   607→            ))}
   608→          </div>
   609→
   610→          {/* Gift card input */}
   611→          {isGiftCard && (
   612→            <div className="space-y-2">
   613→              <div className="flex gap-2">
   614→                <div className="relative flex-1">
   615→                  <Gift className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   616→                  <input type="text" value={giftCardNumber} onChange={(e) => setGiftCardNumber(e.target.value)}
   617→                    placeholder="Gift card number…"
   618→                    className="w-full border border-border rounded-lg pl-8 pr-2 py-2 text-[12px] focus:outline-none focus:border-ink" />
   619→                </div>
   620→                <button onClick={checkGiftCard} className="px-3 py-2 border border-border rounded-lg text-[12px] font-medium hover:bg-muted cursor-pointer">Check</button>
   621→              </div>
   622→              {giftCardBalance != null && <p className="text-[11px] text-success">Balance: ${giftCardBalance.toFixed(2)}</p>}
   623→            </div>
   624→          )}
   625→
   626→          {/* Check reference */}
   627→          {isCheck && (
   628→            <input type="text" value={checkRef} onChange={(e) => setCheckRef(e.target.value)}
   629→              placeholder="Check # / reference…"
   630→              className="w-full border border-border rounded-lg px-3 py-2 text-[12px] focus:outline-none focus:border-ink" />
   631→          )}
   632→
   633→          {/* Amount + tip */}
   634→          <div className="grid grid-cols-2 gap-3">
   635→            <div>
   636→              <label className="block text-[10px] font-bold uppercase tracking-wider text-muted-foreground mb-1">Amount</label>
   637→              <div className="relative">
   638→                <DollarSign className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   639→                <input type="number" step="0.01" value={amount} onChange={(e) => setAmount(e.target.value)}
   640→                  className="w-full border border-border rounded-lg pl-8 pr-2 py-2 text-[14px] focus:outline-none focus:border-ink" />
   641→              </div>
   642→            </div>
   643→            <div>
   644→              <label className="block text-[10px] font-bold uppercase tracking-wider text-muted-foreground mb-1">Tip</label>
   645→              <div className="relative">
   646→                <DollarSign className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   647→                <input type="number" step="0.01" value={tip} onChange={(e) => setTip(e.target.value)}
   648→                  className="w-full border border-border rounded-lg pl-8 pr-2 py-2 text-[14px] focus:outline-none focus:border-ink" />
   649→              </div>
   650→            </div>
   651→          </div>
   652→
   653→          {/* Change due (cash/check) */}
   654→          {(isCash || isCheck) && changeDue > 0 && (
   655→            <div className="bg-success/10 border border-success/20 rounded-lg p-3 text-center">
   656→              <p className="text-[11px] text-muted-foreground uppercase">Change Due</p>
   657→              <p className="text-[20px] font-bold text-success">${changeDue.toFixed(2)}</p>
   658→              <p className="text-[9px] text-muted-foreground mt-1">Cash drawer will open automatically</p>
   659→            </div>
   660→          )}
   661→
   662→          {/* Quick cash buttons */}
   663→          {(isCash || isCheck) && (
   664→            <div className="grid grid-cols-5 gap-1.5">
   665→              {[20, 50, 100, 'Exact', 'Next'].map((q) => (
   666→                <button key={q} onClick={() => {
   667→                  if (q === 'Exact') setAmount(remaining.toFixed(2));
   668→                  else if (q === 'Next') { const n = Math.ceil(remaining / 20) * 20; setAmount(n.toFixed(2)); }
   669→                  else setAmount(String(q));
   670→                }} className="py-1.5 rounded-lg border border-border text-[11px] font-medium hover:bg-muted cursor-pointer">
   671→                  {q === 'Exact' || q === 'Next' ? q : `$${q}`}
   672→                </button>
   673→              ))}
   674→            </div>
   675→          )}
   676→
   677→          {error && <p className="text-[12px] text-destructive text-center">{error}</p>}
   678→        </div>
   679→
   680→        {/* Actions */}
   681→        <div className="border-t border-border p-4 flex gap-2">
   682→          {remaining > 0.01 && (
   683→            <button onClick={addPayment} className="flex-1 py-2.5 border border-border rounded-lg text-[13px] font-semibold hover:bg-muted cursor-pointer">Add Payment</button>
   684→          )}
   685→          <button
   686→            onClick={() => onComplete(payments.length > 0 ? payments : [{ paymentMethodId: selectedMethod, amount: Number(amount) || total, tipAmount: Number(tip) || 0, giftCardNumber: isGiftCard ? giftCardNumber : undefined, checkReference: isCheck ? checkRef : undefined }])}
   687→            disabled={busy || !selectedMethod}
   688→            className="flex-1 py-2.5 bg-ink text-white rounded-lg text-[13px] font-semibold hover:opacity-90 disabled:opacity-60 cursor-pointer">
   689→            {busy ? 'Processing…' : remaining > 0.01 ? 'Complete (Split)' : 'Complete Sale'}
   690→          </button>
   691→        </div>
   692→      </div>
   693→    </div>
   694→  );
   695→}

   697→// ---- Quick Add Item — create a new catalog item from the register ----
   698→function QuickAddItem({ onClose, onCreated }: {
   699→  onClose: () => void;
   700→  onCreated: (item: { id: string; name: string; price: number }) => void;
   701→}) {
   702→  const [name, setName] = useState('');
   703→  const [price, setPrice] = useState('');
   704→  const [ecoEnabled, setEcoEnabled] = useState(false);
   705→  const [busy, setBusy] = useState(false);
   706→  const [error, setError] = useState<string | null>(null);
   707→
   708→  const create = async () => {
   709→    if (!name.trim() || !price) { setError('Name and price required'); return; }
   710→    setBusy(true); setError(null);
   711→    try {
   712→      const res = await fetch('/api/admin/products', {
   713→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   714→        body: JSON.stringify({
   715→          name: name.trim(),
   716→          base_price: Number(price),
   717→          pos_enabled: true,
   718→          ecommerce_enabled: ecoEnabled,
   719→          short_description: 'Quick-add from POS register',
   720→          inventory_count: 0,
   721→        }),
   722→      });
   723→      const data = await res.json();
   724→      if (!res.ok) { setError(data.error || 'Failed'); setBusy(false); return; }
   725→      onCreated({ id: data.product.id, name: data.product.name, price: data.product.priceCents / 100 });
   726→    } catch { setError('Failed'); setBusy(false); }
   727→  };
   728→
   729→  return (
   730→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-foreground/90 p-4">
   731→      <div className="w-full max-w-md bg-card rounded-2xl border border-border shadow-2xl p-6 space-y-4">
   732→        <div className="flex items-center justify-between">
   733→          <h3 className="text-[16px] font-semibold flex items-center gap-2">
   734→            <Plus className="size-4" /> Quick Add Item
   735→          </h3>
   736→          <button onClick={onClose} className="p-1 rounded-lg text-muted-foreground hover:text-foreground cursor-pointer">
   737→            <X className="size-5" />
   738→          </button>
   739→        </div>
   740→        <p className="text-[12px] text-muted-foreground">
   741→          Create a new item to sell right now. It'll be added to the cart + the catalog.
   742→        </p>
   743→        <div className="space-y-3">
   744→          <div>
   745→            <label className="block text-[11px] font-medium text-foreground mb-1">Item Name</label>
   746→            <input
   747→              type="text" value={name} onChange={(e) => setName(e.target.value)}
   748→              placeholder="e.g. Oatmeal Shampoo (new arrival)"
   749→              className="w-full border border-border rounded-lg px-3 py-2 text-[13px] focus:outline-none focus:border-ink"
   750→              autoFocus
   751→            />
   752→          </div>
   753→          <div>
   754→            <label className="block text-[11px] font-medium text-foreground mb-1">Price ($)</label>
   755→            <div className="relative">
   756→              <DollarSign className="absolute left-2.5 top-1/2 -translate-y-1/2 size-3.5 text-muted-foreground" />
   757→              <input
   758→                type="number" step="0.01" value={price} onChange={(e) => setPrice(e.target.value)}
   759→                placeholder="0.00"
   760→                className="w-full border border-border rounded-lg pl-8 pr-3 py-2 text-[13px] focus:outline-none focus:border-ink"
   761→              />
   762→            </div>
   763→          </div>
   764→          <label className="flex items-center gap-2 text-[12px] text-muted-foreground cursor-pointer">
   765→            <input
   766→              type="checkbox" checked={ecoEnabled} onChange={(e) => setEcoEnabled(e.target.checked)}
   767→              className="rounded border-border"
   768→            />
   769→            Also sell on the website (storefront)
   770→          </label>
   771→        </div>
   772→        {error && <p className="text-[12px] text-destructive text-center">{error}</p>}
   773→        <div className="flex gap-2">
   774→          <button onClick={onClose} className="flex-1 py-2.5 border border-border rounded-lg text-[13px] font-semibold hover:bg-muted cursor-pointer">
   775→            Cancel
   776→          </button>
   777→          <button onClick={create} disabled={busy}
   778→            className="flex-1 py-2.5 bg-ink text-white rounded-lg text-[13px] font-semibold hover:opacity-90 disabled:opacity-60 cursor-pointer">
   779→            {busy ? 'Creating…' : 'Add to Cart'}
   780→          </button>
   781→        </div>
   782→      </div>
   783→    </div>
   784→  );
   785→}
