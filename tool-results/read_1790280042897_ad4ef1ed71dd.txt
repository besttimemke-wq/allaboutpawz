    12→function iso(value: Date | string | null | undefined): string {
    13→  if (!value) return "";
    14→  return typeof value === "string" ? value : value.toISOString();
    15→}

    17→// Convert a Supabase row (timestamps arrive as ISO strings) into the shape
    18→// our callers expect. The LMS data layer reads from the enterprise `lms.*`
    19→// schema (UUIDs + snake_case + tenant_id), so each helper below maps a row
    20→// from `lms.courses` / `lms.enrollments` / etc. back to the CourseRecord /
    21→// Companion shape the classroom UI was authored against.
    22→type Row = Record<string, unknown>;
    23→function asRow<T = Row>(r: unknown): T {
    24→  return (r ?? {}) as T;
    25→}

    36→// Stable integer ID derived from a UUID — the classroom UI uses integer IDs
    37→// for course lookup; lms.courses.id is a UUID. We derive a deterministic
    38→// 31-bit int from the UUID's first 8 hex chars so the same course always
    39→// maps to the same int within a session.
    40→function uuidToInt(uuid: string | null | undefined): number {
    41→  if (!uuid) return 0;
    42→  const head = uuid.replace(/[^0-9a-f]/gi, "").slice(0, 8);
    43→  return parseInt(head || "0", 16) % 0x7fffffff;
    44→}

    66→function mapCourseFromLms(c: LmsCourseRow): CourseRecord {
    67→  const meta = (c.metadata && typeof c.metadata === "object" ? c.metadata : {}) as Record<string, unknown>;
    68→  // Companion resolution order:
    69→  //   1. lms.courses.metadata.companion (if a generated companion was persisted)
    70→  //   2. companionForPathway(c.code) — the curriculum.ts authored companions
    71→  //      keyed by pathway code (IPDG, PDT, ACA, PPS, CAT, PPC, ...)
    72→  //   3. Minimal fallback (title only, empty sections)
    73→  let companion: Companion | null = null;
    74→  const companionRaw = meta.companion;
    75→  if (typeof companionRaw === "string") {
    76→    try { companion = JSON.parse(companionRaw) as Companion; } catch { companion = null; }
    77→  } else if (companionRaw && typeof companionRaw === "object") {
    78→    companion = companionRaw as Companion;
    79→  }
    80→  if (!companion && c.code) {
    81→    try { companion = companionForPathway(c.code); } catch { companion = null; }
    82→  }
    83→  const fallback: Companion = {
    84→    title: c.title, subtitle: "", overview: "", alignment: { state: "TN", grade: "9-12", area: c.category || "Grooming", statute: "", authority: "", note: "" },
    85→    learningObjectives: [], sections: [], independentPractice: [], appliedProject: { title: "", brief: "", deliverables: [] },
    86→    glossary: [], familyNote: "", sources: [],
    87→  };
    88→  return {
    89→    id: uuidToInt(c.id),
    90→    state: (meta.state as string) || companion?.alignment?.state || "TN",
    91→    area: c.category || (meta.area as string) || companion?.alignment?.area || "Grooming",
    92→    statute: (meta.statute as string) || companion?.alignment?.statute || "",
    93→    grade: (meta.grade as string) || companion?.alignment?.grade || "9-12",
    94→    title: c.title,
    95→    companion: companion ?? fallback,
    96→    model: (meta.model as string) || "gemini-1.5-flash",
    97→    createdAt: iso(c.created_at),
    98→  };
    99→}

   101→export async function saveCourse(
   102→  ownerId: string,
   103→  selection: { state: string; area: string; statute: string; grade: string },
   104→  companion: Companion,
   105→  model: string,
   106→) {
   107→  // Courses in the enterprise schema are tenant-scoped, not visitor-scoped.
   108→  // Store the original Prisma-shape fields in lms.courses.metadata so they
   109→  // survive round-trips. slug derived from title; code derived from title.
   110→  const slug = companion.title.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "").slice(0, 60);
   111→  const code = (companion.title.split(/\s+/).map((w: string) => w[0]?.toUpperCase() ?? "").join("").slice(0, 4) || "CRS");
   112→  const meta = JSON.stringify({ state: selection.state, area: selection.area, statute: selection.statute, grade: selection.grade, companion, model, ownerId });
   113→  const rows = await pgQuery<LmsCourseRow>(
   114→    `INSERT INTO lms.courses (tenant_id, code, title, slug, description, course_type, delivery_modes, difficulty_level, is_published, sort_order, metadata)
   115→     VALUES ($1, $2, $3, $4, $5, 'course', ARRAY['self_paced']::text[], 'beginner', true, 999, $6::jsonb)
   116→     RETURNING id, tenant_id, code, title, slug, description, long_description, category, tags, is_published, sort_order, metadata, created_at, total_clock_hours, difficulty_level`,
   117→    [TENANT_ID, code, companion.title, slug, companion.title, meta],
   118→  );
   119→  if (rows.length === 0) return null;
   120→  return mapCourseFromLms(rows[0]);
   121→}

   123→export async function listCourses(ownerId: string): Promise<CourseRecord[]> {
   124→  // The classroom canvas shows the LEARNER'S ENROLLED courses — not the
   125→  // catalog. The catalog (all 15 published) lives at /learn
   126→  // via courses-data.ts; the classroom queries the learner's active
   127→  // enrollments joined to lms.courses for the rich companion data.
   128→  //
   129→  // ownerId is the visitor string ("demo-avery" in this build). The lms.*
   130→  // schema uses UUID user_ids FK'd to auth.users. Until real auth lands,
   131→  // every classroom request maps to the seeded demo learner UUID.
   132→  void ownerId;
   133→  const rows = await pgQuery<LmsCourseRow>(
   134→    `SELECT c.id, c.tenant_id, c.code, c.title, c.slug, c.description, c.long_description, c.category, c.tags, c.is_published, c.sort_order, c.metadata, c.created_at, c.total_clock_hours, c.difficulty_level
   135→     FROM lms.enrollments e
   136→     JOIN lms.courses c ON c.id = e.course_id AND c.tenant_id = e.tenant_id
   137→     WHERE e.learner_user_id = $1 AND e.status = 'active'
   138→     ORDER BY c.sort_order ASC NULLS LAST, c.title ASC
   139→     LIMIT 50`,
   140→    [DEMO_LEARNER_UUID],
   141→  );
   142→  return rows.map(mapCourseFromLms);

   166→// The LMS schema ties tutor messages to a teaching SESSION (ai_teaching_sessions),
   167→// not directly to a course. This helper finds the active session for the demo
   168→// learner + course, creating one if none exists. Returns the session UUID.
   169→async function getOrCreateSession(courseId: number): Promise<string | null> {
   170→  // Find the course UUID from the integer ID
   171→  const courses = await pgQuery<{ id: string }>(
   172→    `SELECT c.id FROM lms.enrollments e
   173→     JOIN lms.courses c ON c.id = e.course_id AND c.tenant_id = e.tenant_id
   174→     WHERE e.learner_user_id = $1 AND e.status = 'active'`,
   175→    [DEMO_LEARNER_UUID],
   176→  );
   177→  const course = courses.find((r) => uuidToInt(r.id) === courseId);
   178→  if (!course) return null;
   179→
   180→  // Find an active teaching session for this learner + course
   181→  const sessions = await pgQuery<{ id: string }>(
   182→    `SELECT id FROM lms.ai_teaching_sessions
   183→     WHERE learner_user_id = $1 AND course_id = $2 AND session_status = 'active'
   184→     ORDER BY created_at DESC LIMIT 1`,
   185→    [DEMO_LEARNER_UUID, course.id],
   186→  );
   187→  if (sessions.length > 0) return sessions[0].id;
   188→
   189→  // No active session — create one
   190→  const inserted = await pgQuery<{ id: string }>(
   191→    `INSERT INTO lms.ai_teaching_sessions
   192→       (id, tenant_id, learner_user_id, course_id, session_status, started_at, delivery_mode)
   193→     VALUES (gen_random_uuid(), $1, $2, $3, 'active', now(), 'ai_guided')
   194→     RETURNING id`,
   195→    [TENANT_ID, DEMO_LEARNER_UUID, course.id],
   196→  );
   197→  return inserted.length > 0 ? inserted[0].id : null;
   198→}

   200→export async function listMessages(ownerId: string, courseId: number) {
   201→  void ownerId;
   202→  const sessionId = await getOrCreateSession(courseId);
   203→  if (!sessionId) return [];
   204→  const rows = await pgQuery<{
   205→    id: string;
   206→    message_role: string;
   207→    message_content: string;
   208→    created_at: string;
   209→  }>(
   210→    `SELECT id, message_role, message_content, created_at
   211→     FROM lms.ai_tutor_messages
   212→     WHERE session_id = $1
   213→     ORDER BY created_at ASC
   214→     LIMIT 100`,
   215→    [sessionId],
   216→  );
   217→  return rows.map((r, i) => ({
   218→    id: i + 1,
   219→    role: (r.message_role === "ai" ? "professor" : r.message_role) as "learner" | "professor",
   220→    content: r.message_content,
   221→    createdAt: iso(r.created_at),
   222→  }));
   223→}

   784→// Deterministic synthetic day ID — stable per owner+course. The persisted
   785→// learningDay table was part of the dropped public.* schema; until the
   786→// day-state migration to lms.* lands, we synthesize a day ID so the
   787→// classroom can open a day and enable the Professor + tools.
   788→function syntheticDayId(ownerId: string, courseId: number): number {
   789→  const str = ownerId + ":" + courseId;
   790→  let h = 0;
   791→  for (let i = 0; i < str.length; i++) {
   792→    h = ((h << 5) - h + str.charCodeAt(i)) | 0;
   793→  }
   794→  return Math.abs(h) % 0x7fffffff || 1;
   795→}

  1040→    .from("learningDay")
  1041→    .select("id, courseId")
  1042→    .eq("ownerId", ownerId)
  1043→    .order("id", { ascending: false })

  1045→  if (error) {
  1046→    console.error("[db] listInstructorDaySnapshots failed:", error.message);
  1047→    return [];
  1048→  }
  1049→  // Keep only the latest day per course (mirrors the original GROUP BY MAX(id)).
  1050→  const seen = new Set<number>();
  1051→  const latest = (rows ?? []).filter((r) => {
  1052→    const courseId = asRow<{ courseId: number }>(r).courseId;
  1053→    if (seen.has(courseId)) return false;
  1054→    seen.add(courseId);
  1055→    return true;
  1056→  });
  1057→  // Fetch course titles for each unique course id (no FK relationship to lean
  1058→  // on for the embedded PostgREST join — we do it in JS instead).
  1059→  const courseIds = latest.map((r) => asRow<{ courseId: number }>(r).courseId);
  1060→  const courseInfoMap = new Map<number, { area: string; title: string }>();
  1061→  if (courseIds.length > 0) {
  1062→    const { data: courseRows, error: courseErr } = await supabase
  1063→      .from("course")
  1064→      .select("id, area, title")
  1065→      .in("id", courseIds);
  1066→    if (courseErr) console.error("[db] listInstructorDaySnapshots course fetch:", courseErr.message);
  1067→    for (const c of courseRows ?? []) {
  1068→      const r = asRow<{ id: number; area: string; title: string }>(c);
  1069→      courseInfoMap.set(r.id, { area: r.area, title: r.title });
  1070→    }
  1071→  }
  1072→  const snapshots = await Promise.all(
  1073→    latest.map(async (r) => {
  1074→      const courseId = asRow<{ courseId: number }>(r).courseId;
  1075→      const courseInfo = courseInfoMap.get(courseId);
  1076→      const snapshot = await getLearningDaySnapshot(ownerId, courseId);
  1077→      return {
  1078→        ...snapshot,
  1079→        courseTitle: courseInfo?.title ?? "",
  1080→        courseArea: courseInfo?.area ?? "",
  1081→        learnerName: "Avery Johnson",
  1082→      };
  1083→    }),
  1084→  );
  1085→  return snapshots.slice(0, 30);
  1086→}
