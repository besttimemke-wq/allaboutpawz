     9→function iso(value: Date | string | null | undefined): string {
    10→  if (!value) return "";
    11→  return typeof value === "string" ? value : value.toISOString();
    12→}

    29→function mapCourse(c: CourseRow): CourseRecord {
    30→  return {
    31→    id: c.id,
    32→    state: c.state,
    33→    area: c.area,
    34→    statute: c.statute,
    35→    grade: c.grade,
    36→    title: c.title,
    37→    companion: JSON.parse(c.companionJson) as Companion,
    38→    model: c.model,
    39→    createdAt: iso(c.createdAt),
    40→  };

    44→  ownerId: string,
    45→  selection: { state: string; area: string; statute: string; grade: string },
    46→  companion: Companion,
    47→  model: string,
    48→) {

    55→      grade: selection.grade,
    56→      title: companion.title,
    57→      companionJson: JSON.stringify(companion),

   433→async function courseFirstItem(ownerId: string, courseId: number) {
   434→  const course = await getCourse(ownerId, courseId);
   435→  const first = course?.companion.sections?.[0];
   436→  return (
   437→    first?.checks?.[0] ||
   438→    `Explain the central idea of ${first?.title || course?.title || "this lesson"} in your own words.`
   439→  );
   440→}

   459→export async function openLearningDay(
   460→  ownerId: string,
   461→  courseId: number,
   462→  mode: "SPRINT" | "SHIFT" | "FULL_DAY",
   463→) {
   464→  const active = await prisma.learningDay.findFirst({
   465→    where: { ownerId, courseId, closedAt: null },
   466→    orderBy: { id: "desc" },
   467→  });
   468→  if (active) return active.id;
   469→
   470→  const duration = mode === "FULL_DAY" ? 480 : mode === "SHIFT" ? 120 : 45;
   471→  const now = new Date();
   472→  const item = await courseFirstItem(ownerId, courseId);
   473→  const row = await prisma.learningDay.create({
   474→    data: {
   475→      ownerId,
   476→      courseId,
   477→      mode,
   478→      state: "CHECK_IN",
   479→      currentLesson: 0,
   480→      currentItem: item,
   481→      cycleStep: 0,
   482→      openedAt: now,
   483→      scheduledCloseAt: new Date(now.getTime() + duration * 60000),
   484→      updatedAt: now,
   485→    },
   486→  });
   487→  await appendDayEvent(ownerId, row.id, "DAY_OPENED", { mode, durationMinutes: duration });
   488→  return row.id;
   489→}

   491→function resolveDate(next: string | null | undefined, current: Date | null): Date | null {
   492→  if (next === undefined) return current;
   493→  if (next === null) return null;
   494→  return new Date(next);
   495→}

   497→export async function commandLearningDay(ownerId: string, dayId: number, command: DayCommand) {
   498→  const current = await prisma.learningDay.findFirst({ where: { ownerId, id: dayId } });
   499→  if (!current) throw new Error("Learning Day not found.");
   500→
   501→  const state = command.state ?? current.state;
   502→  const lesson = command.currentLesson ?? current.currentLesson;
   503→  const item = command.currentItem ?? current.currentItem;
   504→  const cycle = command.cycleStep ?? current.cycleStep;
   505→  const sentiment = command.sentiment ?? current.sentiment;
   506→  const breakEnds = resolveDate(command.breakEndsAt, current.breakEndsAt);
   507→  const prior = command.priorState === undefined ? current.priorState : command.priorState;
   508→  const workKind = command.activeWorkKind ?? (current.activeWorkKind || "LESSON");
   509→  const workKey = command.activeWorkKey === undefined ? current.activeWorkKey : command.activeWorkKey;
   510→  const workTitle =
   511→    command.activeWorkTitle === undefined ? current.activeWorkTitle : command.activeWorkTitle;
   512→
   513→  await prisma.learningDay.update({
   514→    where: { id: dayId },
   515→    data: {
   516→      state,
   517→      currentLesson: lesson,
   518→      currentItem: item,
   519→      cycleStep: cycle,
   520→      sentiment,
   521→      breakEndsAt: breakEnds,
   522→      priorState: prior,
   523→      activeWorkKind: workKind,
   524→      activeWorkKey: workKey,
   525→      activeWorkTitle: workTitle,
   526→      version: { increment: 1 },
   527→      updatedAt: new Date(),
   528→    },
   529→  });
   530→  await appendDayEvent(ownerId, dayId, command.eventType, {
   531→    fromState: current.state,
   532→    toState: state,
   533→    ...command.payload,
   534→  });
   535→}

   581→  courseId: number,

   592→    });
   593→    return existing.id;
   594→  }
   595→  const row = await prisma.humanNeedQueue.create({
   596→    data: {
   597→      ownerId,
   598→      dayId,
   599→      courseId,
   600→      reason,
   601→      status: "OPEN",
   602→      contextJson: JSON.stringify(context),
   603→      createdAt: new Date().toISOString(),
   604→    },
   605→  });
   606→  return row.id;

   694→export async function getLearningDaySnapshot(ownerId: string, courseId: number) {
   695→  const row = await prisma.learningDay.findFirst({
   696→    where: { ownerId, courseId },
   697→    orderBy: { id: "desc" },
   698→  });
   699→  if (!row) return null;
   700→
   701→  const [eventRows, attemptRows, queueRows] = await Promise.all([
   702→    prisma.learningDayEvent.findMany({
   703→      where: { ownerId, dayId: row.id },
   704→      orderBy: { id: "asc" },
   705→    }),
   706→    prisma.learningAttempt.findMany({
   707→      where: { ownerId, dayId: row.id },
   708→      orderBy: { id: "asc" },
   709→    }),
   710→    prisma.humanNeedQueue.findMany({
   711→      where: { ownerId, dayId: row.id },
   712→      orderBy: { id: "desc" },
   713→    }),
   714→  ]);
   715→
   716→  const events = eventRows.map((e) => ({
   717→    id: e.id,
   718→    eventType: e.eventType,
   719→    payload: JSON.parse(e.payloadJson),
   720→    createdAt: iso(e.createdAt),
   721→  }));
   722→  const attempts = attemptRows.map((a) => ({
   723→    id: a.id,
   724→    lessonIndex: a.lessonIndex,
   725→    item: a.itemText,
   726→    attemptNo: a.attemptNo,
   727→    response: a.response,
   728→    score: a.score,
   729→    passed: Boolean(a.passed),
   730→    stage: a.stage,
   731→    misconception: a.misconception,
   732→    feedback: a.feedback,
   733→    evidenceSummary: a.evidenceSummary,
   734→    workKind: a.workKind,
   735→    workKey: a.workKey,
   736→    createdAt: iso(a.createdAt),
   737→  }));
   738→  const queue = queueRows.map((q) => ({
   739→    id: q.id,
   740→    reason: q.reason,
   741→    status: q.status,
   742→    context: JSON.parse(q.contextJson),
   743→    createdAt: iso(q.createdAt),
   744→    resolvedAt: q.resolvedAt ? iso(q.resolvedAt) : null,
   745→  }));
   746→
   747→  const passedLessons = new Set(
   748→    attempts.filter((a) => a.passed && a.workKind === "LESSON").map((a) => a.lessonIndex),
   749→  );
   750→  const openedAt = new Date(row.openedAt).getTime();
   751→  const endedAt = row.closedAt ? new Date(row.closedAt).getTime() : Date.now();
   752→  let cursor = openedAt;
   753→  let activeMs = 0;
   754→  let paused = false;
   755→  for (const event of events) {
   756→    const at = new Date(event.createdAt).getTime();
   757→    if (event.eventType === "BREAK_STARTED") {
   758→      if (!paused) activeMs += Math.max(0, at - cursor);
   759→      paused = true;
   760→    } else if (event.eventType === "BREAK_ENDED") {
   761→      paused = false;
   762→      cursor = at;
   763→    }
   764→  }
   765→  if (!paused) activeMs += Math.max(0, endedAt - cursor);
   766→
   767→  return {
   768→    day: mapDay(row),
   769→    events,
   770→    attempts,
   771→    queue,
   772→    metrics: {
   773→      activeMinutes: Math.floor(activeMs / 60000),
   774→      demonstratedLessons: passedLessons.size,
   775→      evidenceCount: attempts.filter((a) => a.passed).length,
   776→    },
   777→  };
   778→}

   810→function todayInChicago(): string {
   811→  const parts = new Intl.DateTimeFormat("en-US", {
   812→    timeZone: "America/Chicago",
   813→    year: "numeric",
   814→    month: "2-digit",
   815→    day: "2-digit",
   816→  }).formatToParts(new Date());
   817→  const pick = (type: string) => parts.find((p) => p.type === type)?.value || "";
   818→  return `${pick("year")}-${pick("month")}-${pick("day")}`;
   819→}

   821→export async function getSchoolSnapshot(ownerId: string) {
   822→  const today = todayInChicago();
   823→
   824→  const enrollmentRows = await prisma.courseEnrollment.findMany({
   825→    where: { ownerId },
   826→    include: { course: true },
   827→  });
   828→  const priorityRank: Record<string, number> = { High: 1, Medium: 2, Low: 3 };
   829→  const enrollments = enrollmentRows
   830→    .map((e) => ({ e, course: e.course }))
   831→    .sort((a, b) => {
   832→      const rank = (priorityRank[a.e.priority] ?? 3) - (priorityRank[b.e.priority] ?? 3);
   833→      if (rank !== 0) return rank;
   834→      return a.course.title.localeCompare(b.course.title);
   835→    });
   836→
   837→  const hasTodayBlock = await prisma.schoolScheduleBlock.findFirst({
   838→    where: { ownerId, schoolDate: today },
   839→    select: { id: true },
   840→  });
   841→  const latest = await prisma.schoolScheduleBlock.findFirst({
   842→    where: { ownerId },
   843→    orderBy: { schoolDate: "desc" },
   844→    select: { schoolDate: true },
   845→  });
   846→  const displayedDate = hasTodayBlock ? today : latest?.schoolDate || today;
   847→
   848→  const [scheduleRows, gradeRows, meetingRows] = await Promise.all([
   849→    prisma.schoolScheduleBlock.findMany({
   850→      where: { ownerId, schoolDate: displayedDate },
   851→      orderBy: { sequence: "asc" },
   852→      include: { course: { select: { area: true, title: true } } },
   853→    }),
   854→    prisma.gradebookEntry.findMany({
   855→      where: { ownerId },
   856→      include: { course: { select: { title: true } } },
   857→    }),
   858→    prisma.classroomMeeting.findMany({
   859→      where: { ownerId },
   860→      orderBy: { startsAt: "asc" },
   861→      include: { course: { select: { title: true } } },
   862→    }),
   863→  ]);
   864→
   865→  const schedule = scheduleRows.map((s) => ({
   866→    id: s.id,
   867→    courseId: s.courseId,
   868→    blockType: s.blockType,
   869→    title: s.title,
   870→    startsAt: s.startsAt,
   871→    endsAt: s.endsAt,
   872→    status: s.status,
   873→    deficiencyFocus: s.deficiencyFocus,
   874→    sequence: s.sequence,
   875→    area: s.course?.area,
   876→    courseTitle: s.course?.title,
   877→  }));
   878→
   879→  const grades = gradeRows
   880→    .map((g) => ({
   881→      id: g.id,
   882→      courseId: g.courseId,
   883→      courseTitle: g.course.title,
   884→      title: g.title,
   885→      category: g.category,
   886→      score: g.score,
   887→      possible: g.possible,
   888→      status: g.status,
   889→      feedback: g.feedback,
   890→      gradedAt: g.gradedAt ? iso(g.gradedAt) : null,
   891→    }))
   892→    .sort((a, b) => {
   893→      const ga = a.gradedAt ?? "9999";
   894→      const gb = b.gradedAt ?? "9999";
   895→      if (ga !== gb) return gb.localeCompare(ga);
   896→      return b.id - a.id;
   897→    });
   898→
   899→  const meetings = meetingRows.map((m) => ({
   900→    id: m.id,
   901→    courseId: m.courseId,
   902→    title: m.title,
   903→    startsAt: m.startsAt,
   904→    endsAt: m.endsAt,
   905→    room: m.room,
   906→    status: m.status,
   907→    courseTitle: m.course?.title,
   908→  }));
   909→
   910→  const assignmentStates = await listAssignmentStates(ownerId);
   911→  const submissions = assignmentStates
   912→    .filter((item) => ["Submitted", "Completed"].includes(item.status))
   913→    .map((item) => {
   914→      const enrollment = enrollments.find((en) => en.course.id === item.courseId);
   915→      const companion = enrollment ? (JSON.parse(enrollment.course.companionJson) as Companion) : null;
   916→      const work = [
   917→        ...(companion?.independentPractice || []).map((title, index) => ({
   918→          key: `practice-${index}`,
   919→          title,
   920→          kind: "Assignment",
   921→        })),
   922→        ...(companion?.appliedProject
   923→          ? [{ key: "project", title: companion.appliedProject.title, kind: "Project" }]
   924→          : []),
   925→        ...(companion?.sections || []).flatMap((section, lessonIndex) =>
   926→          (section.checks || []).map((title, index) => ({
   927→            key: `check-${lessonIndex}-${index}`,
   928→            title,
   929→            kind: "Quiz",
   930→          })),
   931→        ),
   932→      ].find((entry) => entry.key === item.assignmentKey);
   933→      return {
   934→        ...item,
   935→        title: work?.title || item.assignmentKey,
   936→        kind: work?.kind || "Assignment",
   937→        courseTitle: enrollment?.course.title || "Course",
   938→      };
   939→    });
   940→
   941→  return {
   942→    schoolDate: displayedDate,
   943→    currentDate: today,
   944→    isHistoricalSchedule: displayedDate !== today,
   945→    enrollments: enrollments.map(({ e, course }) => {
   946→      const companion = JSON.parse(course.companionJson) as Companion;
   947→      return {
   948→        courseId: course.id,
   949→        title: course.title,
   950→        area: course.area,
   951→        grade: course.grade,
   952→        state: course.state,
   953→        level: e.level,
   954→        deficiencyFocus: e.deficiencyFocus,
   955→        priority: e.priority,
   956→        enrolledAt: iso(e.enrolledAt),
   957→        lessonCount: (companion.sections || []).length,
   958→        objectiveCount: (companion.learningObjectives || []).length,
   959→        nextLesson: companion.sections?.[0]?.title || course.title,
   960→      };
   961→    }),
   962→    schedule,
   963→    grades,
   964→    submissions,
   965→    meetings,
   966→    inbox: (await listWorkspaceMessages(ownerId)).slice(-50).reverse(),
   967→    resources: await listWorkspaceFiles(ownerId),
   968→    events: await listWorkspaceEvents(ownerId),
   969→  };
   970→}
