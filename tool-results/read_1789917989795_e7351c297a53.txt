    27→// Track code mapping to program id
    28→const TRACK_CODE_TO_PROGRAM_ID: Record<string, string> = {

    30→  PDT: 'pdt',
    31→  ACA: 'aca',

    53→      // Pathway filter
    54→      if (selectedPathway !== 'all') {
    55→        if (selectedPathway === 'grooming' && !['IPDG', 'CAT', 'ACA'].includes(m.trackCode)) return false;
    56→        if (selectedPathway === 'training' && m.trackCode !== 'PDT') return false;
    57→        if (selectedPathway === 'pet-care' && !['PPS', 'ACA', 'PPC'].includes(m.trackCode)) return false;
    58→        if (selectedPathway === 'business' && !['PPC', 'BIZ', 'PERS'].includes(m.trackCode)) return false;
    59→      }
    60→
    61→      // Course Level filter
    62→      if (selectedLevel !== 'all') {
    63→        if (m.level !== selectedLevel) return false;
    64→      }
    65→
    66→      // Search Query filter
    67→      if (searchQuery.trim()) {
    68→        const q = searchQuery.toLowerCase().trim();
    69→        const matchesCode = m.code.toLowerCase().includes(q);
    70→        const matchesTitle = m.title.toLowerCase().includes(q);
    71→        const matchesDesc = m.description.toLowerCase().includes(q);
    72→        const matchesTrack = m.trackTitle.toLowerCase().includes(q) || m.trackCode.toLowerCase().includes(q);
    73→        const matchesCategory = m.category.toLowerCase().includes(q);
    74→        const matchesSafety = (q.includes('safety') || q.includes('gate')) && !!m.safetyGate;
    75→        const matchesPracticum = q.includes('practicum') && !!m.practicum;
    76→
    77→        if (!matchesCode && !matchesTitle && !matchesDesc && !matchesTrack && !matchesCategory && !matchesSafety && !matchesPracticum) {
    78→          return false;
    79→        }
    80→      }
    81→
    82→      return true;
    83→    });
    84→  }, [selectedPathway, selectedLevel, searchQuery]);
    85→

    87→  const programMatchingModulesMap = useMemo(() => {
    88→    const map: Record<string, CatalogModule[]> = {};
    89→    if (!searchQuery.trim() && selectedLevel === 'all') return map;
    90→
    91→    for (const m of filteredModules) {
    92→      const progId = TRACK_CODE_TO_PROGRAM_ID[m.trackCode];
    93→      if (progId) {
    94→        if (!map[progId]) map[progId] = [];
    95→        map[progId].push(m);
    96→      }
    97→    }
    98→    return map;
    99→  }, [filteredModules, searchQuery, selectedLevel]);
   100→
   101→  // 3. Filtered & Sorted Programs

   104→      // Pathway filter
   105→      if (selectedPathway !== 'all') {
   106→        if (selectedPathway === 'grooming' && !['ipdg', 'cat', 'aca'].includes(p.id)) return false;
   107→        if (selectedPathway === 'training' && p.id !== 'pdt') return false;
   108→        if (selectedPathway === 'pet-care' && !['pps', 'aca', 'ppc'].includes(p.id)) return false;
   109→        if (selectedPathway === 'business' && p.id !== 'ppc') return false;
   110→      }
   111→
   112→      // Duration filter
   113→      if (selectedDuration === 'short' && p.totalWeeks > 16) return false;

   115→
   116→      // Course Level filter
   117→      if (selectedLevel !== 'all') {
   118→        const hasLevelModule = ALL_CATALOG_MODULES.some(
   119→          (m) => TRACK_CODE_TO_PROGRAM_ID[m.trackCode] === p.id && m.level === selectedLevel
   120→        );
   121→        if (!hasLevelModule) return false;
   122→      }
   123→
   124→      // Search Query filter
   125→      if (searchQuery.trim()) {
   126→        const q = searchQuery.toLowerCase().trim();
   127→        const matchesTitle = p.title.toLowerCase().includes(q);
   128→        const matchesCode = p.code.toLowerCase().includes(q);
   129→        const matchesSubtitle = p.subtitle.toLowerCase().includes(q);
   130→        const matchesOverview = p.overviewParagraphs.some((para) => para.toLowerCase().includes(q));
   131→        const matchesTopics = p.terms.some((t) => t.topics.some((top) => top.toLowerCase().includes(q)));
   132→        const matchesHighlights = p.terms.some((t) =>
   133→          t.courseHighlights.some(
   134→            (ch) =>
   135→              ch.code.toLowerCase().includes(q) ||
   136→              ch.title.toLowerCase().includes(q) ||
   137→              ch.description.toLowerCase().includes(q)
   138→          )
   139→        );
   140→        const hasMatchedModules = (programMatchingModulesMap[p.id]?.length ?? 0) > 0;
   141→        const matchesSafetyQuery = (q.includes('safety') || q.includes('gate')) && p.terms.some((t) =>
   142→          t.courseHighlights.some((ch) => ch.title.toLowerCase().includes('safety gate'))
   143→        );
   144→
   145→        if (
   146→          !matchesTitle &&
   147→          !matchesCode &&
   148→          !matchesSubtitle &&
   149→          !matchesOverview &&
   150→          !matchesTopics &&
   151→          !matchesHighlights &&
   152→          !hasMatchedModules &&
   153→          !matchesSafetyQuery
   154→        ) {
   155→          return false;
   156→        }
   157→      }
   158→
   159→      return true;
   160→    });
   161→
   162→    // Sorting
   163→    return [...list].sort((a, b) => {
   164→      if (sortBy === 'hours') return b.totalClockHours - a.totalClockHours;
   165→      if (sortBy === 'weeks') return b.totalWeeks - a.totalWeeks;
   166→      if (sortBy === 'name') return a.title.localeCompare(b.title);
   167→      return 0; // 'featured' keeps original curriculum order
   168→    });
   169→  }, [selectedPathway, selectedDuration, selectedLevel, searchQuery, sortBy, programMatchingModulesMap]);

   173→    selectedLevel !== 'all' ||
   174→    selectedDuration !== 'all' ||
   175→    searchQuery.trim() !== '';

   179→    setSelectedLevel('all');
   180→    setSelectedDuration('all');
   181→    setSearchQuery('');
   182→  };
