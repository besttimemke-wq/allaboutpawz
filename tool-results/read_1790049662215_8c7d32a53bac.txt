    35→// Google OAuth redirect-URI activation (LIVE verification — no stale lists)
    36→// ---------------------------------------------------------------------------
    37→// Ground truth for "is this origin activated" lives in the owner's Google
    38→// client, NOT in this codebase. The previous hardcoded origin list was a
    39→// second gate on top of Google's own: an origin the owner had already
    40→// registered still bounced, because the list only changes when the code does.
    41→// That double gate is gone. /api/auth/google/start now verifies the redirect
    42→// URI against Google's own authorize endpoint at click time: with prompt=none,
    43→// a REGISTERED redirect URI answers 302 back to that URI
    44→// (error=interaction_required / a code); an UNREGISTERED one answers 302 to
    45→// accounts.google.com/signin/oauth/error (redirect_uri_mismatch). Results are
    46→// cached in memory — positive 5 min, negative 60 s — so a URI registered in
    47→// the Google console takes effect within a minute, with no code or env change.
    48→// On a probe network failure the check fails OPEN (Google renders its own
    49→// error — the app never blocks a working flow).
    50→//
    51→// GOOGLE_REGISTERED_ORIGINS (comma-separated) is still honored as an
    52→// additional skip-the-probe allowlist for deployments that want zero
    53→// outbound calls to Google on the sign-in path.
    54→

    81→const uriCache = new Map<string, { ok: boolean; at: number }>();
    82→
    83→/** Live check: does the Google client have this exact redirect URI registered? */
    84→export async function redirectUriRegistered(redirectUri: string): Promise<boolean> {
    85→  const allowlisted = (process.env.GOOGLE_REGISTERED_ORIGINS || "")
    86→    .split(",")
    87→    .map((s) => s.trim().replace(/\/$/, ""))
    88→    .filter(Boolean)
    89→    .some((o) => redirectUri === googleCallbackUri(o));
    90→  if (allowlisted) return true;
    91→
    92→  const hit = uriCache.get(redirectUri);
    93→  const ttl = hit?.ok ? 5 * 60 * 1000 : 60 * 1000;
    94→  if (hit && Date.now() - hit.at < ttl) return hit.ok;
    95→
    96→  let ok = true; // fail-open — let Google itself render any error
    97→  try {
    98→    const probeUrl =
    99→      "https://accounts.google.com/o/oauth2/v2/auth?" +
   100→      new URLSearchParams({
   101→        client_id: process.env.GOOGLE_CLIENT_ID || "",
   102→        redirect_uri: redirectUri,
   103→        response_type: "code",
   104→        scope: "openid email profile",
   105→        state: "check",
   106→        prompt: "none",
   107→      }).toString();
   108→    const res = await fetch(probeUrl, { redirect: "manual", signal: AbortSignal.timeout(5000) });
   109→    const loc = res.headers.get("location") || "";
   110→    if (res.status === 302 && loc) ok = loc.startsWith(redirectUri);
   111→  } catch {
   112→    ok = true;
   113→  }
   114→  uriCache.set(redirectUri, { ok, at: Date.now() });
   115→  return ok;
   116→}
   117→
   118→let relayCheck: { ok: boolean; status: number | null; at: number } | null = null;
   119→
   120→/** Live check: is the production deployment running relay-capable callback
   121→ *  code (GET {relay}/api/auth/google/callback?probe=relay → {relay:true})?
   122→ *  Preview-origin flows only relay when this is true, so users are never
   123→ *  silently signed into the wrong deployment. Fails CLOSED — an unreachable
   124→ *  probe never routes anyone into a dead end.
   125→ *
   126→ *  Redirects are FOLLOWED: the registered redirect URI is the apex
   127→ *  (https://aapawz.com/...) while the deployment may sit behind an apex→www
   128→ *  308 — a manual-redirect probe would report "not capable" forever even
   129→ *  after the build is deployed. The JSON + content-type check on the final
   130→ *  response keeps the probe honest (an HTML catch-all page never passes). */
   131→export async function productionRelayStatus(): Promise<{ ok: boolean; status: number | null }> {
   132→  const hit = relayCheck;
   133→  const ttl = hit?.ok ? 5 * 60 * 1000 : 60 * 1000;
   134→  if (hit && Date.now() - hit.at < ttl) return { ok: hit.ok, status: hit.status };
   135→
   136→  let ok = false;
   137→  let status: number | null = null;
   138→  try {
   139→    const res = await fetch(`${productionRelayCallbackUri()}?probe=relay`, {
   140→      redirect: "follow",
   141→      headers: { accept: "application/json" },
   142→      signal: AbortSignal.timeout(5000),
   143→    });
   144→    status = res.status;
   145→    if (res.status === 200 && (res.headers.get("content-type") || "").includes("application/json")) {
   146→      const body = await res.json().catch(() => null);
   147→      ok = Boolean(body && (body as any).relay === true);
   148→    }
   149→  } catch (e: any) {
   150→    ok = false;
   151→    status = null;
   152→  }
   153→  relayCheck = { ok, status, at: Date.now() };
   154→  return { ok, status };
   155→}

   157→/** Boolean convenience wrapper around productionRelayStatus(). */
   158→export async function productionRelayCapable(): Promise<boolean> {
   159→  return (await productionRelayStatus()).ok;
   160→}

   172→export function newBrowserBinding(): { value: string; hash: string } {
   173→  const value = randomBytes(32).toString("base64url");
   174→  return { value, hash: createHash("sha256").update(value).digest("hex") };
   175→}

   178→  if (!value) return null;
   179→  return createHash("sha256").update(value).digest("hex");

   233→/**
   234→ * Resolves the portal identity of an authenticated Supabase auth user:
   235→ * platform_admins → tenant_memberships → staff → portal_customer_accounts →
   236→ * customers.userId → user metadata fallback.
   237→ */
   238→export async function resolvePortalUser(authUserId: string): Promise<ResolvedPortalUser | null> {
   239→  const admin = getSupabaseAdmin();
   240→  if (!admin) return null;
   241→
   242→  const { data: authUserData } = await admin.auth.admin.getUserById(authUserId);
   243→  const authUser: any = authUserData?.user;
   244→  const email = (authUser?.email || "").toLowerCase();
   245→  if (!email) return null;
   246→
   247→  const nameFromMeta =
   248→    authUser?.user_metadata?.full_name ||
   249→    authUser?.user_metadata?.name ||
   250→    (authUser?.user_metadata as any)?.firstName ||
   251→    email.split("@")[0];
   252→  const avatarUrl = authUser?.user_metadata?.avatar_url || authUser?.user_metadata?.picture;
   253→
   254→  // 0. Owner-declared admins (ADMIN_EMAILS) — first and final word. The owner's
   255→  //    account is an admin regardless of table state; no provisioning row can
   256→  //    demote it, and an account created directly in Supabase by the owner
   257→  //    still resolves as admin.
   258→  const declaredAdmins = (process.env.ADMIN_EMAILS || "")
   259→    .split(",")
   260→    .map((e) => e.trim().toLowerCase())
   261→    .filter(Boolean);
   262→  if (declaredAdmins.includes(email)) {
   263→    // Prefer the salon's own staff record for the display name (best-effort).
   264→    let name = nameFromMeta;
   265→    try {
   266→      const { data: staffRow } = await admin.from("staff").select("name").ilike("email", email).limit(1);
   267→      if (staffRow && staffRow.length > 0 && (staffRow[0] as any).name) name = (staffRow[0] as any).name;
   268→    } catch { /* best-effort */ }
   269→    return { authUserId, email, name, role: "admin", avatarUrl, scope: "admin", membershipRole: "owner" };
   270→  }
   271→
   272→  // 1. Platform admins (cross-tenant super admins)
   273→  try {
   274→    const { data: platformAdmins } = await admin.from("platform_admins").select("user_id").eq("user_id", authUserId).limit(1);
   275→    if (platformAdmins && platformAdmins.length > 0) {
   276→      return { authUserId, email, name: nameFromMeta, role: "admin", avatarUrl, scope: "admin", membershipRole: "platform_admin" };
   277→    }
   278→  } catch { /* table may not exist */ }
   279→
   280→  // 2. Tenant memberships (admin / employee scopes)
   281→  try {
   282→    const { data: memberships } = await admin
   283→      .from("tenant_memberships")
   284→      .select("role, tenant_id, status")
   285→      .eq("user_id", authUserId);
   286→    const active = (memberships || []).find((m: any) => m.status !== "suspended") || (memberships || [])[0];
   287→    if (active) {
   288→      const membershipRole = String(active.role || "").toLowerCase();
   289→      if (ADMIN_ROLES.includes(membershipRole)) {
   290→        return { authUserId, email, name: nameFromMeta, role: "admin", avatarUrl, scope: "admin", membershipRole };
   291→      }
   292→      if (FRONTDESK_ROLES.includes(membershipRole)) {
   293→        return {
   294→          authUserId, email, name: nameFromMeta, role: "admin", avatarUrl, scope: "employee",
   295→          stationName: "Front Desk — Intake & Concierge", membershipRole,
   296→        };
   297→      }
   298→      if (GROOMER_ROLES.includes(membershipRole)) {
   299→        return { authUserId, email, name: nameFromMeta, role: "groomer", avatarUrl, scope: "employee", membershipRole };
   300→      }
   301→      // Unknown staff role → employee/groomer portal by default
   302→      return { authUserId, email, name: nameFromMeta, role: "groomer", avatarUrl, scope: "employee", membershipRole };
   303→    }
   304→  } catch { /* table may not exist */ }
   305→
   306→  // 3. Staff table (fallback employee signal). EMAIL IS THE ASSOCIATION KEY:
   307→  // when a staff row exists for this email but its userId is NULL or points
   308→  // at a stale auth id (user re-created in auth.users, row provisioned
   309→  // earlier), the email match associates the sign-in with the salon record
   310→  // and the link self-heals — the sign-in NEVER bounces for a provisioning
   311→  // row that is merely out of date.
   312→  try {
   313→    const { data: staff } = await admin.from("staff").select("name, role").eq("userId", authUserId).limit(1);
   314→    let staffRow = staff && staff.length > 0 ? (staff[0] as any) : null;
   315→    if (!staffRow && email) {
   316→      const { data: byEmail } = await admin
   317→        .from("staff")
   318→        .select("id, name, role, userId")
   319→        .ilike("email", email)
   320→        .not("email", "is", null)
   321→        .limit(1);
   322→      if (byEmail && byEmail.length > 0) {
   323→        staffRow = byEmail[0] as any;
   324→        // Self-heal: bind the row to the signing-in auth user (stale or NULL).
   325→        if (staffRow.userId !== authUserId) {
   326→          try {
   327→            await admin.from("staff").update({ userId: authUserId }).eq("id", staffRow.id);
   328→          } catch { /* heal is best-effort; resolution proceeds */ }
   329→        }
   330→      }
   331→    }
   332→    if (staffRow) {
   333→      const staffRole = String(staffRow.role || "").toLowerCase();
   334→      const name = staffRow.name || nameFromMeta;
   335→      if (FRONTDESK_ROLES.includes(staffRole)) {
   336→        return { authUserId, email, name, role: "admin", avatarUrl, scope: "employee", stationName: "Front Desk — Intake & Concierge", membershipRole: staffRole };
   337→      }
   338→      if (ADMIN_ROLES.includes(staffRole)) {
   339→        return { authUserId, email, name, role: "admin", avatarUrl, scope: "admin", membershipRole: staffRole };
   340→      }
   341→      return { authUserId, email, name, role: "groomer", avatarUrl, scope: "employee", membershipRole: staffRole };
   342→    }
   343→  } catch { /* table may not exist */ }
   344→
   345→  // 4. Customer portal accounts
   346→  try {
   347→    const { data: accounts } = await admin
   348→      .from("portal_customer_accounts")
   349→      .select("auth_user_id, status")
   350→      .eq("auth_user_id", authUserId)
   351→      .limit(1);
   352→    if (accounts && accounts.length > 0) {
   353→      return { authUserId, email, name: nameFromMeta, role: "customer", avatarUrl, scope: "customer", membershipRole: "customer" };
   354→    }
   355→  } catch { /* table may not exist */ }
   356→
   357→  // 5. Customers linked by userId — with the same email-based self-heal as
   358→  // staff: an existing customer row whose userId is NULL (created at
   359→  // checkout/walk-in before the account existed) is associated by email and
   360→  // the link is repaired on this sign-in.
   361→  try {
   362→    const { data: customers } = await admin.from("customers").select("firstName, lastName").eq("userId", authUserId).limit(1);
   363→    let customerRow = customers && customers.length > 0 ? (customers[0] as any) : null;
   364→    if (!customerRow && email) {
   365→      const { data: byEmail } = await admin
   366→        .from("customers")
   367→        .select("id, firstName, lastName, userId")
   368→        .ilike("email", email)
   369→        .not("email", "is", null)
   370→        .limit(1);
   371→      if (byEmail && byEmail.length > 0) {
   372→        customerRow = byEmail[0] as any;
   373→        if (!customerRow.userId) {
   374→          try {
   375→            await admin.from("customers").update({ userId: authUserId }).eq("id", customerRow.id);
   376→          } catch { /* heal is best-effort */ }
   377→        }
   378→      }
   379→    }
   380→    if (customerRow) {
   381→      const c = customerRow;
   382→      const name = `${c.firstName || ""} ${c.lastName || ""}`.trim() || nameFromMeta;
   383→      return { authUserId, email, name, role: "customer", avatarUrl, scope: "customer", membershipRole: "customer" };
   384→    }
   385→  } catch { /* table may not exist */ }
   386→
   387→  // 6. Metadata fallback (never grants staff access — customers only)
   388→  const metaRole = String(authUser?.user_metadata?.role || "").toLowerCase();
   389→  if (metaRole === "customer") {
   390→    return { authUserId, email, name: nameFromMeta, role: "customer", avatarUrl, scope: "customer", membershipRole: "customer" };
   391→  }
   392→  // Auth user with NO provisioning records at all → treat as customer only
   393→  // for the customer/lms doors; staff doors reject them explicitly.
   394→  return { authUserId, email, name: nameFromMeta, role: "customer", avatarUrl, scope: "customer", membershipRole: "unprovisioned" };
   395→}

   407→/** The ORIGINAL repo's routing rule (Serviceportals /api/auth/google):
   408→ *  "DB is source of truth" — the destination comes from the resolved salon
   409→ *  identity, never from the door, a URL param, or anything the user picked. */
   410→export function autoDestination(user: ResolvedPortalUser): string {
   411→  // Front desk employees get their OWN portal now (not the admin dashboard).
   412→  if (FRONTDESK_ROLES.includes(String(user.membershipRole || ""))) return "/frontdesk/dashboard";
   413→  if (user.scope === "admin") return "/admin/dashboard";
   414→  if (user.scope === "employee") return "/groomer/dashboard";
   415→  return "/customer/dashboard";

   418→export function validatePortalAccess(portal: PortalId, user: ResolvedPortalUser): PortalValidationResult {
   419→  const def = PORTALS[portal];
   420→
   421→  switch (portal) {
   422→    case "admin": {
   423→      if (user.scope === "admin" && user.role === "admin" && user.membershipRole !== "front_desk") {
   424→        return { ok: true, user, redirectTo: def.destination };
   425→      }
   426→      if (FRONTDESK_ROLES.includes(String(user.membershipRole || ""))) {
   427→        return { ok: false, error: "This account is registered for Front Desk access. Use the Front Desk sign-in page." };
   428→      }
   429→      if (user.scope === "employee") {
   430→        return { ok: false, error: "No admin account found for this email. Contact your admin." };
   431→      }
   432→      return { ok: false, error: "This email is registered as a customer account. Use the Customer sign-in page." };
   433→    }
   434→    case "groomer": {
   435→      if (user.scope === "employee" && user.role === "groomer") {
   436→        return { ok: true, user, redirectTo: def.destination };
   437→      }
   438→      if (FRONTDESK_ROLES.includes(String(user.membershipRole || ""))) {
   439→        return { ok: false, error: "This account is registered for Front Desk access. Use the Front Desk sign-in page." };
   440→      }
   441→      if (user.scope === "admin") {
   442→        return { ok: false, error: "This email is an admin account. Use the Admin sign-in page." };
   443→      }
   444→      return { ok: false, error: "No groomer account found for this email. Contact your admin." };
   445→    }
   446→    case "frontdesk": {
   447→      if (user.scope === "employee" && FRONTDESK_ROLES.includes(String(user.membershipRole || ""))) {
   448→        return { ok: true, user, redirectTo: def.destination };
   449→      }
   450→      if (user.scope === "employee") {
   451→        return { ok: false, error: "This account is registered for the Groomer portal. Use the Groomer sign-in page." };
   452→      }
   453→      if (user.scope === "admin") {
   454→        return { ok: false, error: "This email is an admin account. Use the Admin sign-in page." };
   455→      }
   456→      return { ok: false, error: "No front desk account found for this email. Contact your admin." };
   457→    }
   458→    case "customer": {
   459→      if (user.role === "customer") {
   460→        return { ok: true, user, redirectTo: def.destination };
   461→      }
   462→      return { ok: false, error: "This email is a staff account. Use the staff sign-in pages." };
   463→    }
   464→    case "lms": {
   465→      // LMS is self-serve within managed accounts — customers AND staff land
   466→      // on the Learning Center dashboard. Each signed-in role gets its own
   467→      // sidebar identity inside the /learn portal.
   468→      if (user.role === "customer") return { ok: true, user, redirectTo: def.destination };
   469→      if (user.scope === "employee" && user.role === "groomer") return { ok: true, user, redirectTo: def.destination };
   470→      if (FRONTDESK_ROLES.includes(String(user.membershipRole || ""))) return { ok: true, user, redirectTo: def.destination };
   471→      if (user.scope === "admin") return { ok: true, user, redirectTo: def.destination };
   472→      return { ok: false, error: "No account found for this email." };
   473→    }
   474→  }

   507→export function signSession(user: ResolvedPortalUser): string {
   508→  const payload: SessionPayload = {
   509→    sub: user.authUserId,
   510→    email: user.email,
   511→    name: user.name,
   512→    role: user.role,
   513→    stationName: user.stationName,
   514→    avatarUrl: user.avatarUrl,
   515→    scope: user.scope,
   516→    membershipRole: user.membershipRole,
   517→    exp: Math.floor(Date.now() / 1000) + SESSION_TTL_SECONDS,
   518→  };
   519→  const body = b64url(JSON.stringify(payload));
   520→  const mac = createHmac("sha256", secret()).update(body).digest("base64url");
   521→  return `${body}.${mac}`;
   522→}

   524→export function verifySessionToken(token: string | undefined | null): SessionPayload | null {
   525→  if (!token) return null;
   526→  const parts = token.split(".");
   527→  if (parts.length !== 2) return null;
   528→  const [body, mac] = parts;
   529→  const expected = createHmac("sha256", secret()).update(body).digest("base64url");
   530→  try {
   531→    const a = Buffer.from(mac);
   532→    const b = Buffer.from(expected);
   533→    if (a.length !== b.length || !timingSafeEqual(a, b)) return null;
   534→  } catch {
   535→    return null;
   536→  }
   537→  try {
   538→    const payload = JSON.parse(Buffer.from(body, "base64url").toString("utf-8")) as SessionPayload;
   539→    if (!payload?.sub || !payload?.exp) return null;
   540→    if (payload.exp < Math.floor(Date.now() / 1000)) return null;
   541→    return payload;
   542→  } catch {
   543→    return null;
   544→  }
   545→}

   571→// ---------------------------------------------------------------------------
   572→// Apex ↔ www cookie continuity (THE production sign-in killer, fixed)
   573→// ---------------------------------------------------------------------------
   574→// The registered Google redirect URI is the APEX (https://aapawz.com/...),
   575→// but the platform 308-redirects the apex to https://www.aapawz.com — so a
   576→// flow that starts on one host always finishes on the other. Host-only
   577→// cookies (no Domain attribute) set by the initiator on the apex were
   578→// INVISIBLE to the callback on www: the browser-binding guard failed, and
   579→// the callback silently bounced the owner back to the login page after they
   580→// had authorized at Google (oauth_states row consumed, no google_sub linked —
   581→// 2026-09-17 11:53 UTC production incident).
   582→//
   583→// Fix: on the salon's own hosts, issue auth cookies with Domain=.aapawz.com
   584→// so they are shared across apex and www. On every other host (localhost,
   585→// sandbox previews) NO domain is emitted — a Domain that is not a suffix of
   586→// the current host is rejected by the browser, so this stays safe everywhere.
   587→const SITE_COOKIE_DOMAIN = ".aapawz.com";
   588→
   589→export function cookieDomainForHost(reqHost?: string | null): { domain?: string } {
   590→  const host = (reqHost || "").split(":")[0].trim().toLowerCase();
   591→  if (host === "aapawz.com" || host.endsWith(".aapawz.com")) {
   592→    return { domain: SITE_COOKIE_DOMAIN };
   593→  }
   594→  return {};
   595→}

   597→/** Best-effort public host of the current request, for cookie domainning.
   598→ *  Prefers X-Forwarded-Host (the real browser host behind any gateway),
   599→ *  falls back to the URL-level host. */
   600→export function requestHost(req: { headers: { get(name: string): string | null }; nextUrl?: { host: string } }): string | null {
   601→  const xfh = req.headers.get("x-forwarded-host");
   602→  if (xfh) return xfh.split(",")[0].trim().toLowerCase() || null;
   603→  if (req.nextUrl?.host) return req.nextUrl.host.toLowerCase();
   604→  return null;
   605→}

   624→  returnOrigin?: string | null;

   626→   *  (login-CSRF binding — see OAUTH_BROWSER_COOKIE). */
   627→  browserHash?: string | null;

   655→/** Creates a single-use OAuth state row in public.oauth_states (service-role). */
   656→export async function createOAuthState(
   657→  portal: PortalId,
   658→  redirectTo: string,
   659→  redirectUri?: string,
   660→  returnOrigin?: string | null,
   661→  browserHash?: string | null,
   662→): Promise<string | null> {
   663→  const admin = getSupabaseAdmin();
   664→  if (!admin) return null;
   665→  // Housekeeping: states are single-use and short-lived; sweep rows that have
   666→  // been dead for over a day so the table never grows unboundedly. Best-effort.
   667→  try {
   668→    await admin.from("oauth_states").delete().lt("expires_at", new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString());
   669→  } catch { /* hygiene is best-effort */ }
   670→  const nonce = randomBytes(32).toString("base64url");
   671→  const expiresAt = new Date(Date.now() + OAUTH_STATE_TTL_MS).toISOString();
   672→  const { error } = await admin
   673→    .from("oauth_states")
   674→    .insert({ nonce, portal, redirect_to: redirectTo, redirect_uri: redirectUri || null, return_origin: returnOrigin || null, browser_hash: browserHash || null, expires_at: expiresAt, used: false });
   675→  if (error) {
   676→    console.error("[pawz-auth] createOAuthState:", error.message);
   677→    return null;
   678→  }
   679→  return signStateNonce(nonce);
   680→}

   682→/** Maps a raw oauth_states row to the typed shape (no consumption). */
   683→function mapStateRow(row: any): OAuthStateRow | null {
   684→  const portal = row.portal as PortalId;
   685→  if (!PORTALS[portal]) return null;
   686→  return {
   687→    nonce: row.nonce,
   688→    portal,
   689→    redirectTo: row.redirect_to || PORTALS[portal].destination,
   690→    redirectUri: row.redirect_uri || null,
   691→    returnOrigin: row.return_origin || null,
   692→    browserHash: row.browser_hash || null,
   693→    expiresAt: row.expires_at,
   694→    used: Boolean(row.used),
   695→  };
   696→}
   697→
   698→const OAUTH_STATE_COLUMNS = "nonce, portal, redirect_to, redirect_uri, return_origin, browser_hash, expires_at, used";
   699→
   700→/** Verifies the signed state and reads its row WITHOUT consuming it — used
   701→ *  by the callback to decide whether this flow belongs to another origin
   702→ *  (relay) before anything is marked used. */
   703→export async function peekOAuthState(state: string | null | undefined): Promise<OAuthStateRow | null> {
   704→  const nonce = verifyStateSignature(state);
   705→  if (!nonce) return null;
   706→  const admin = getSupabaseAdmin();
   707→  if (!admin) return null;
   708→  const { data: rows, error } = await admin
   709→    .from("oauth_states")
   710→    .select(OAUTH_STATE_COLUMNS)
   711→    .eq("nonce", nonce)
   712→    .limit(1);
   713→  if (error || !rows || rows.length === 0) return null;
   714→  const row = mapStateRow(rows[0]);
   715→  if (!row) return null;
   716→  if (row.used) return null;
   717→  if (new Date(row.expiresAt).getTime() < Date.now()) return null;
   718→  return row;
   719→}
   720→
   721→/** Verifies + consumes a state value. Returns the row, or null on any failure. */
   722→export async function consumeOAuthState(state: string | null | undefined): Promise<OAuthStateRow | null> {
   723→  const nonce = verifyStateSignature(state);
   724→  if (!nonce) return null;
   725→  const admin = getSupabaseAdmin();
   726→  if (!admin) return null;
   727→
   728→  const { data: rows, error } = await admin
   729→    .from("oauth_states")
   730→    .select(OAUTH_STATE_COLUMNS)
   731→    .eq("nonce", nonce)
   732→    .limit(1);
   733→  if (error || !rows || rows.length === 0) return null;
   734→
   735→  const row = mapStateRow(rows[0]);
   736→  if (!row) return null;
   737→  if (row.used) return null; // replay — reject
   738→  if (new Date(row.expiresAt).getTime() < Date.now()) return null; // expired
   739→
   740→  // Mark used immediately (single-use). If the update fails, reject.
   741→  const { error: updateError } = await admin.from("oauth_states").update({ used: true }).eq("nonce", nonce).eq("used", false);
   742→  if (updateError) {
   743→    // Raced with another callback — treat as replay.
   744→    return null;
   745→  }
   746→
   747→  return { ...row, used: true };
   748→}

   779→export async function exchangeGoogleCode(code: string, redirectUri: string): Promise<{ profile: GoogleProfile | null; error?: string }> {
   780→  try {
   781→    const tokenRes = await fetch("https://oauth2.googleapis.com/token", {
   782→      method: "POST",
   783→      headers: { "Content-Type": "application/x-www-form-urlencoded" },
   784→      body: new URLSearchParams({
   785→        code,
   786→        client_id: process.env.GOOGLE_CLIENT_ID || "",
   787→        client_secret: process.env.GOOGLE_CLIENT_SECRET || "",
   788→        redirect_uri: redirectUri,
   789→        grant_type: "authorization_code",
   790→      }),
   791→    });
   792→    const tokenData: any = await tokenRes.json();
   793→    if (!tokenRes.ok) {
   794→      return { profile: null, error: tokenData?.error_description || tokenData?.error || "Google token exchange failed." };
   795→    }
   796→    // The id_token came directly from Google's token endpoint over TLS —
   797→    // decoding the payload (no local signature check needed) is standard.
   798→    const idToken = tokenData.id_token as string | undefined;
   799→    if (!idToken) return { profile: null, error: "Google response did not include an identity token." };
   800→    const payload = JSON.parse(Buffer.from(idToken.split(".")[1], "base64url").toString("utf-8")) as any;
   801→    if (!payload?.sub || !payload?.email) {
   802→      return { profile: null, error: "Google identity token is missing an email." };
   803→    }
   804→    return {
   805→      profile: {
   806→        sub: String(payload.sub),
   807→        email: String(payload.email).toLowerCase(),
   808→        name: payload.name,
   809→        picture: payload.picture,
   810→        emailVerified: payload.email_verified !== false,
   811→      },
   812→    };
   813→  } catch (e: any) {
   814→    return { profile: null, error: e?.message || "Google token exchange failed." };
   815→  }
   816→}

   818→export async function findAuthUserByEmail(admin: SupabaseClient, email: string) {
   819→  // Primary path: O(1) RPC against auth.users (migration 0008). This is the
   820→  // reliable lookup — the paginated listUsers call below is capped at 4000
   821→  // users and is rate-limited, so on a busy tenant it silently returned null
   822→  // for an existing user and the callback bounced them to the door.
   823→  try {
   824→    const { data, error } = await admin
   825→      .rpc("get_auth_user_by_email", { p_email: email });
   826→    if (error) {
   827→      console.error("[pawz-auth] get_auth_user_by_email RPC error:", error.message);
   828→    } else if (data && data.length > 0) {
   829→      const row = data[0] as any;
   830→      return {
   831→        id: row.id,
   832→        email: row.email,
   833→        user_metadata: row.raw_user_meta_data || {},
   834→        identities: [],
   835→        created_at: row.created_at,
   836→        last_sign_in_at: row.last_sign_in_at,
   837→      } as any;
   838→    }
   839→    // RPC resolved cleanly with no rows → genuinely no user. Skip the
   840→    // paginated fallback (it would return null too).
   841→    if (!error) return null;
   842→  } catch (e: any) {
   843→    console.error("[pawz-auth] get_auth_user_by_email RPC threw:", e?.message || e);
   844→  }
   845→
   846→  // Fallback: paginated listUsers (only if the RPC is missing — e.g. the
   847→  // migration hasn't been applied yet on this environment). Capped at 4000
   848→  // users; the RPC above is the real fix.
   849→  let page = 1;
   850→  for (;;) {
   851→    const { data, error } = await admin.auth.admin.listUsers({ page, perPage: 200 });
   852→    if (error) throw error;
   853→    const users = data?.users || [];
   854→    const found = users.find((u: any) => (u.email || "").toLowerCase() === email.toLowerCase());
   855→    if (found) return found as any;
   856→    if (users.length < 200 || page > 20) return null;
   857→    page += 1;
   858→  }
   859→}
