    12→    12→function iso(value: Date | string | null | undefined): string {
    13→    13→  if (!value) return "";
    14→    14→  return typeof value === "string" ? value : value.toISOString();
    15→    15→}

    17→    17→// Convert a Supabase row (timestamps arrive as ISO strings) into the shape
    18→    18→// our callers expect. The LMS data layer reads from the enterprise `lms.*`
    19→    19→// schema (UUIDs + snake_case + tenant_id), so each helper below maps a row
    20→    20→// from `lms.courses` / `lms.enrollments` / etc. back to the CourseRecord /

    22→    22→type Row = Record<string, unknown>;
    23→    23→function asRow<T = Row>(r: unknown): T {
    24→    24→  return (r ?? {}) as T;

    36→    36→// Stable integer ID derived from a UUID — the classroom UI uses integer IDs
    37→    37→// for course lookup; lms.courses.id is a UUID. We derive a deterministic
    38→    38→// 31-bit int from the UUID's first 8 hex chars so the same course always
    39→    39→// maps to the same int within a session.
    40→    40→function uuidToInt(uuid: string | null | undefined): number {
    41→    41→  if (!uuid) return 0;
    42→    42→  const head = uuid.replace(/[^0-9a-f]/gi, "").slice(0, 8);
    43→    43→  return parseInt(head || "0", 16) % 0x7fffffff;
    44→    44→}

    66→    66→function mapCourseFromLms(c: LmsCourseRow): CourseRecord {
    67→    67→  const meta = (c.metadata && typeof c.metadata === "object" ? c.metadata : {}) as Record<string, unknown>;
    68→    68→  // Companion resolution order:
    69→    69→  //   1. lms.courses.metadata.companion (if a generated companion was persisted)
    70→    70→  //   2. companionForPathway(c.code) — the curriculum.ts authored companions
    71→    71→  //      keyed by pathway code (IPDG, PDT, ACA, PPS, CAT, PPC, ...)
    72→    72→  //   3. Minimal fallback (title only, empty sections)
    73→    73→  let companion: Companion | null = null;
    74→    74→  const companionRaw = meta.companion;
    75→    75→  if (typeof companionRaw === "string") {
    76→    76→    try { companion = JSON.parse(companionRaw) as Companion; } catch { companion = null; }
    77→    77→  } else if (companionRaw && typeof companionRaw === "object") {
    78→    78→    companion = companionRaw as Companion;
    79→    79→  }
    80→    80→  if (!companion && c.code) {
    81→    81→    try { companion = companionForPathway(c.code); } catch { companion = null; }
    82→    82→  }
    83→    83→  const fallback: Companion = {
    84→    84→    title: c.title, subtitle: "", overview: "", alignment: { state: "TN", grade: "9-12", area: c.category || "Grooming", statute: "", authority: "", note: "" },
    85→    85→    learningObjectives: [], sections: [], independentPractice: [], appliedProject: { title: "", brief: "", deliverables: [] },
    86→    86→    glossary: [], familyNote: "", sources: [],
    87→    87→  };
    88→    88→  return {
    89→    89→    id: uuidToInt(c.id),
    90→    90→    state: (meta.state as string) || companion?.alignment?.state || "TN",
    91→    91→    area: c.category || (meta.area as string) || companion?.alignment?.area || "Grooming",
    92→    92→    statute: (meta.statute as string) || companion?.alignment?.statute || "",
    93→    93→    grade: (meta.grade as string) || companion?.alignment?.grade || "9-12",
    94→    94→    title: c.title,
    95→    95→    companion: companion ?? fallback,
    96→    96→    model: (meta.model as string) || "gemini-1.5-flash",
    97→    97→    createdAt: iso(c.created_at),
    98→    98→  };
    99→    99→}

   101→   101→export async function saveCourse(
   102→   102→  ownerId: string,
   103→   103→  selection: { state: string; area: string; statute: string; grade: string },
   104→   104→  companion: Companion,
   105→   105→  model: string,
   106→   106→) {
   107→   107→  // Courses in the enterprise schema are tenant-scoped, not visitor-scoped.
   108→   108→  // Store the original Prisma-shape fields in lms.courses.metadata so they
   109→   109→  // survive round-trips. slug derived from title; code derived from title.
   110→   110→  const slug = companion.title.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "").slice(0, 60);
   111→   111→  const code = (companion.title.split(/\s+/).map((w: string) => w[0]?.toUpperCase() ?? "").join("").slice(0, 4) || "CRS");
   112→   112→  const meta = JSON.stringify({ state: selection.state, area: selection.area, statute: selection.statute, grade: selection.grade, companion, model, ownerId });
   113→   113→  const rows = await pgQuery<LmsCourseRow>(
   114→   114→    `INSERT INTO lms.courses (tenant_id, code, title, slug, description, course_type, delivery_modes, difficulty_level, is_published, sort_order, metadata)
   115→   115→     VALUES ($1, $2, $3, $4, $5, 'course', ARRAY['self_paced']::text[], 'beginner', true, 999, $6::jsonb)
   116→   116→     RETURNING id, tenant_id, code, title, slug, description, long_description, category, tags, is_published, sort_order, metadata, created_at, total_clock_hours, difficulty_level`,
   117→   117→    [TENANT_ID, code, companion.title, slug, companion.title, meta],
   118→   118→  );
   119→   119→  if (rows.length === 0) return null;
   120→   120→  return mapCourseFromLms(rows[0]);
   121→   121→}

   166→   166→// The LMS schema ties tutor messages to a teaching SESSION (ai_teaching_sessions),
   167→   167→// not directly to a course. This helper finds the active session for the demo
   168→   168→// learner + course, creating one if none exists. Returns the session UUID.
   169→   169→async function getOrCreateSession(courseId: number): Promise<string | null> {
   170→   170→  // Find the course UUID from the integer ID
   171→   171→  const courses = await pgQuery<{ id: string }>(
   172→   172→    `SELECT c.id FROM lms.enrollments e
   173→   173→     JOIN lms.courses c ON c.id = e.course_id AND c.tenant_id = e.tenant_id
   174→   174→     WHERE e.learner_user_id = $1 AND e.status = 'active'`,
   175→   175→    [DEMO_LEARNER_UUID],
   176→   176→  );
   177→   177→  const course = courses.find((r) => uuidToInt(r.id) === courseId);
   178→   178→  if (!course) return null;
   179→   179→
   180→   180→  // Find an active teaching session for this learner + course
   181→   181→  const sessions = await pgQuery<{ id: string }>(
   182→   182→    `SELECT id FROM lms.ai_teaching_sessions
   183→   183→     WHERE learner_user_id = $1 AND course_id = $2 AND session_status = 'active'
   184→   184→     ORDER BY created_at DESC LIMIT 1`,
   185→   185→    [DEMO_LEARNER_UUID, course.id],
   186→   186→  );
   187→   187→  if (sessions.length > 0) return sessions[0].id;
   188→   188→
   189→   189→  // No active session — create one
   190→   190→  const inserted = await pgQuery<{ id: string }>(
   191→   191→    `INSERT INTO lms.ai_teaching_sessions
   192→   192→       (id, tenant_id, learner_user_id, course_id, session_status, started_at, delivery_mode)
   193→   193→     VALUES (gen_random_uuid(), $1, $2, $3, 'active', now(), 'ai_guided')
   194→   194→     RETURNING id`,
   195→   195→    [TENANT_ID, DEMO_LEARNER_UUID, course.id],
   196→   196→  );
   197→   197→  return inserted.length > 0 ? inserted[0].id : null;
   198→   198→}

   784→   784→// Deterministic synthetic day ID — stable per owner+course. The persisted
   785→   785→// learningDay table was part of the dropped public.* schema; until the
   786→   786→// day-state migration to lms.* lands, we synthesize a day ID so the
   787→   787→// classroom can open a day and enable the Professor + tools.
   788→   788→function syntheticDayId(ownerId: string, courseId: number): number {
   789→   789→  const str = ownerId + ":" + courseId;
   790→   790→  let h = 0;
   791→   791→  for (let i = 0; i < str.length; i++) {
   792→   792→    h = ((h << 5) - h + str.charCodeAt(i)) | 0;
   793→   793→  }
   794→   794→  return Math.abs(h) % 0x7fffffff || 1;
   795→   795→}
