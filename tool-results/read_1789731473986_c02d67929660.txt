   459→   459→export async function openLearningDay(
   460→   460→  ownerId: string,
   461→   461→  courseId: number,
   462→   462→  mode: "SPRINT" | "SHIFT" | "FULL_DAY",
   463→   463→) {
   464→   464→  const active = await prisma.learningDay.findFirst({
   465→   465→    where: { ownerId, courseId, closedAt: null },
   466→   466→    orderBy: { id: "desc" },
   467→   467→  });
   468→   468→  if (active) return active.id;
   469→   469→
   470→   470→  const duration = mode === "FULL_DAY" ? 480 : mode === "SHIFT" ? 120 : 45;
   471→   471→  const now = new Date();
   472→   472→  const item = await courseFirstItem(ownerId, courseId);
   473→   473→  const row = await prisma.learningDay.create({
   474→   474→    data: {
   475→   475→      ownerId,
   476→   476→      courseId,
   477→   477→      mode,
   478→   478→      state: "CHECK_IN",
   479→   479→      currentLesson: 0,
   480→   480→      currentItem: item,
   481→   481→      cycleStep: 0,
   482→   482→      openedAt: now,
   483→   483→      scheduledCloseAt: new Date(now.getTime() + duration * 60000),
   484→   484→      updatedAt: now,
   485→   485→    },
   486→   486→  });
   487→   487→  await appendDayEvent(ownerId, row.id, "DAY_OPENED", { mode, durationMinutes: duration });
   488→   488→  return row.id;
   489→   489→}

   491→   491→function resolveDate(next: string | null | undefined, current: Date | null): Date | null {
   492→   492→  if (next === undefined) return current;
   493→   493→  if (next === null) return null;
   494→   494→  return new Date(next);
   495→   495→}

   497→   497→export async function commandLearningDay(ownerId: string, dayId: number, command: DayCommand) {
   498→   498→  const current = await prisma.learningDay.findFirst({ where: { ownerId, id: dayId } });
   499→   499→  if (!current) throw new Error("Learning Day not found.");
   500→   500→
   501→   501→  const state = command.state ?? current.state;
   502→   502→  const lesson = command.currentLesson ?? current.currentLesson;
   503→   503→  const item = command.currentItem ?? current.currentItem;
   504→   504→  const cycle = command.cycleStep ?? current.cycleStep;
   505→   505→  const sentiment = command.sentiment ?? current.sentiment;
   506→   506→  const breakEnds = resolveDate(command.breakEndsAt, current.breakEndsAt);
   507→   507→  const prior = command.priorState === undefined ? current.priorState : command.priorState;
   508→   508→  const workKind = command.activeWorkKind ?? (current.activeWorkKind || "LESSON");
   509→   509→  const workKey = command.activeWorkKey === undefined ? current.activeWorkKey : command.activeWorkKey;
   510→   510→  const workTitle =
   511→   511→    command.activeWorkTitle === undefined ? current.activeWorkTitle : command.activeWorkTitle;
   512→   512→
   513→   513→  await prisma.learningDay.update({
   514→   514→    where: { id: dayId },
   515→   515→    data: {
   516→   516→      state,
   517→   517→      currentLesson: lesson,
   518→   518→      currentItem: item,
   519→   519→      cycleStep: cycle,

   694→   694→export async function getLearningDaySnapshot(ownerId: string, courseId: number) {
   695→   695→  const row = await prisma.learningDay.findFirst({
   696→   696→    where: { ownerId, courseId },
   697→   697→    orderBy: { id: "desc" },
   698→   698→  });
   699→   699→  if (!row) return null;
   700→   700→
   701→   701→  const [eventRows, attemptRows, queueRows] = await Promise.all([
   702→   702→    prisma.learningDayEvent.findMany({
   703→   703→      where: { ownerId, dayId: row.id },
   704→   704→      orderBy: { id: "asc" },
   705→   705→    }),
   706→   706→    prisma.learningAttempt.findMany({
   707→   707→      where: { ownerId, dayId: row.id },
   708→   708→      orderBy: { id: "asc" },
   709→   709→    }),
   710→   710→    prisma.humanNeedQueue.findMany({
   711→   711→      where: { ownerId, dayId: row.id },
   712→   712→      orderBy: { id: "desc" },
   713→   713→    }),
   714→   714→  ]);
   715→   715→
   716→   716→  const events = eventRows.map((e) => ({
   717→   717→    id: e.id,
   718→   718→    eventType: e.eventType,
   719→   719→    payload: JSON.parse(e.payloadJson),
   720→   720→    createdAt: iso(e.createdAt),
   721→   721→  }));
   722→   722→  const attempts = attemptRows.map((a) => ({
   723→   723→    id: a.id,
   724→   724→    lessonIndex: a.lessonIndex,
   725→   725→    item: a.itemText,
   726→   726→    attemptNo: a.attemptNo,
   727→   727→    response: a.response,
   728→   728→    score: a.score,
   729→   729→    passed: Boolean(a.passed),
   730→   730→    stage: a.stage,
   731→   731→    misconception: a.misconception,
   732→   732→    feedback: a.feedback,
   733→   733→    evidenceSummary: a.evidenceSummary,
   734→   734→    workKind: a.workKind,
   735→   735→    workKey: a.workKey,
   736→   736→    createdAt: iso(a.createdAt),
   737→   737→  }));
   738→   738→  const queue = queueRows.map((q) => ({
   739→   739→    id: q.id,
   740→   740→    reason: q.reason,
   741→   741→    status: q.status,
   742→   742→    context: JSON.parse(q.contextJson),
   743→   743→    createdAt: iso(q.createdAt),
   744→   744→    resolvedAt: q.resolvedAt ? iso(q.resolvedAt) : null,
   745→   745→  }));
   746→   746→
   747→   747→  const passedLessons = new Set(
   748→   748→    attempts.filter((a) => a.passed && a.workKind === "LESSON").map((a) => a.lessonIndex),
   749→   749→  );
   750→   750→  const openedAt = new Date(row.openedAt).getTime();
   751→   751→  const endedAt = row.closedAt ? new Date(row.closedAt).getTime() : Date.now();
   752→   752→  let cursor = openedAt;
   753→   753→  let activeMs = 0;
   754→   754→  let paused = false;
   755→   755→  for (const event of events) {
   756→   756→    const at = new Date(event.createdAt).getTime();
   757→   757→    if (event.eventType === "BREAK_STARTED") {
   758→   758→      if (!paused) activeMs += Math.max(0, at - cursor);
   759→   759→      paused = true;
   760→   760→    } else if (event.eventType === "BREAK_ENDED") {
   761→   761→      paused = false;
   762→   762→      cursor = at;
   763→   763→    }
   764→   764→  }
   765→   765→  if (!paused) activeMs += Math.max(0, endedAt - cursor);
   766→   766→
   767→   767→  return {
   768→   768→    day: mapDay(row),
   769→   769→    events,
   770→   770→    attempts,
   771→   771→    queue,
   772→   772→    metrics: {
   773→   773→      activeMinutes: Math.floor(activeMs / 60000),
   774→   774→      demonstratedLessons: passedLessons.size,
   775→   775→      evidenceCount: attempts.filter((a) => a.passed).length,
   776→   776→    },
   777→   777→  };
   778→   778→}
