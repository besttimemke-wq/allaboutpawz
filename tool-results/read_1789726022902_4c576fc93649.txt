   573→   573→export function getLearningDaySnapshot(ownerId:string,courseId:number) {
   574→   574→  ensureDaySchema();
   575→   575→  const row=db().prepare(`SELECT * FROM learning_days WHERE owner_id=? AND course_id=? ORDER BY id DESC LIMIT 1`)
   576→   576→    .get(ownerId,courseId) as any;
   577→   577→  if(!row) return null;
   578→   578→  const events=(db().prepare(`SELECT id,event_type AS eventType,payload_json AS payload,created_at AS createdAt
   579→   579→    FROM learning_day_events WHERE owner_id=? AND day_id=? ORDER BY id`).all(ownerId,row.id) as any[])
   580→   580→    .map(e=>({...e,payload:JSON.parse(e.payload)}));
   581→   581→  const attempts=(db().prepare(`SELECT id,lesson_index AS lessonIndex,item_text AS item,attempt_no AS attemptNo,
   582→   582→    response,score,passed,stage,misconception,feedback,evidence_summary AS evidenceSummary,
   583→   583→    work_kind AS workKind,work_key AS workKey,created_at AS createdAt
   584→   584→    FROM learning_attempts WHERE owner_id=? AND day_id=? ORDER BY id`).all(ownerId,row.id) as any[])
   585→   585→    .map(a=>({...a,passed:Boolean(a.passed)}));
   586→   586→  const queue=(db().prepare(`SELECT id,reason,status,context_json AS context,created_at AS createdAt,resolved_at AS resolvedAt
   587→   587→    FROM human_need_queue WHERE owner_id=? AND day_id=? ORDER BY id DESC`).all(ownerId,row.id) as any[])
   588→   588→    .map(q=>({...q,context:JSON.parse(q.context)}));
   589→   589→  const passedLessons=new Set(attempts.filter(a=>a.passed&&a.workKind==="LESSON").map(a=>a.lessonIndex));
   590→   590→  const openedAt=new Date(row.opened_at).getTime();
   591→   591→  const endedAt=row.closed_at?new Date(row.closed_at).getTime():Date.now();
   592→   592→  let cursor=openedAt,activeMs=0,paused=false;
   593→   593→  for(const event of events){
   594→   594→    const at=new Date(event.createdAt).getTime();
   595→   595→    if(event.eventType==="BREAK_STARTED"){
   596→   596→      if(!paused)activeMs+=Math.max(0,at-cursor);
   597→   597→      paused=true;
   598→   598→    }else if(event.eventType==="BREAK_ENDED"){
   599→   599→      paused=false;cursor=at;
   600→   600→    }
   601→   601→  }
   602→   602→  if(!paused)activeMs+=Math.max(0,endedAt-cursor);
   603→   603→  return {
   604→   604→    day:mapDay(row),events,attempts,queue,
   605→   605→    metrics:{activeMinutes:Math.floor(activeMs/60000),demonstratedLessons:passedLessons.size,evidenceCount:attempts.filter(a=>a.passed).length},
   606→   606→  };
   607→   607→}

   648→   648→  const todayParts=new Intl.DateTimeFormat("en-US",{timeZone:"America/Chicago",year:"numeric",month:"2-digit",day:"2-digit"}).formatToParts(new Date());
   649→   649→  const pick=(type:string)=>todayParts.find(part=>part.type===type)?.value||"";
   650→   650→  const today=`${pick("year")}-${pick("month")}-${pick("day")}`;

   670→   670→    FROM gradebook_entries g JOIN courses c ON c.id=g.course_id

   689→   689→  });

   693→   693→    isHistoricalSchedule:displayedDate!==today,

   695→   695→      ...item,
   696→   696→      lessonCount:(JSON.parse(companionJson).sections||[]).length,
   697→   697→      objectiveCount:(JSON.parse(companionJson).learningObjectives||[]).length,
   698→   698→      nextLesson:JSON.parse(companionJson).sections?.[0]?.title||item.title,
